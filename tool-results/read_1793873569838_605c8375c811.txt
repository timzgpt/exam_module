   133→   133→                  </span>

   136→   136→            ) : null}

   138→   138→              <img
   139→   139→                src={chartImage}

   141→   141→                className="chart-img w-full"
   142→   142→              />

   144→   144→              <p className="text-2xs text-muted-foreground">{CHART_FALLBACK_TEXT}</p>

   147→   147→              // Terminal legend is rendered in HTML, not in the chart image (§9).
   148→   148→              <div className="mt-1 flex flex-wrap items-center justify-center gap-x-4 gap-y-1 text-2xs text-muted-foreground">

   157→   157→                </span>
   158→   158→              </div>
   159→   159→            ) : null}
   160→   160→            {/* Chart data-table fallback (B0: charts include a table fallback). */}
   161→   161→            <table className="sr-only">
   162→   162→              <caption>Subject totals against the class average and class top</caption>
   163→   163→              <thead>
   164→   164→                <tr>

   173→   173→                  <tr key={row.name}>
   174→   174→                    <th scope="row">{row.name}</th>
   175→   175→                    <td>{fmtTotalMark(row.achieved)}</td>

   189→   189→export default function ReportCardPage() {
   190→   190→  const params = useParams<{ studentId?: string }>();
   191→   191→  const router = useRouter();
   192→   192→  const { cycle } = useCycleContext();
   193→   193→  const student = studentById(params.studentId ?? "") ?? studentById(DEMO_IDS.studentId)!;
   194→   194→
   195→   195→  const result = studentResultOf(student.id);
   196→   196→  const classmates = studentsInClass(student.classId);
   197→   197→  const classSize = classmates.length;
   198→   198→  const attendance = attendanceRecordFor(student.id);
   199→   199→  const comment = commentFor(student.classId, student.id);
   200→   200→  const classTeacherName =
   201→   201→    teacherById(classById(student.classId)?.classTeacherId)?.name ?? "\u2014";
   202→   202→
   203→   203→  const classSubjects = classSubjectsFor(student.classId);
   204→   204→  const scheme = schemeForClass(student.classId);
   205→   205→  const isPrimary = isPrimaryClass(student.classId);
   206→   206→  // Assessment split shown in the table header (25 % monthly / 75 % terminal).
   207→   207→  const caPct = classSubjects[0]?.caMarks ?? 25;
   208→   208→  const examPct = classSubjects[0]?.examMarks ?? 75;
   209→   209→  // Top band of the class scheme — rendered as the filled grade badge.
   210→   210→  const topGrade = scheme.boundaries[0]?.grade;
   211→   211→  const allRows = classSubjects.map((cs) => {
   212→   212→    const subject = subjectById(cs.subjectId)!;
   213→   213→    const comp = getComponents(student.classId, cs.subjectId)[student.id];
   214→   214→    const enrolled = comp?.enrolled ?? true;
   215→   215→    const mark = enrolled ? result?.marks[cs.subjectId] ?? null : null;
   216→   216→    const grade = mark !== null ? gradeFor(mark, scheme) : undefined;
   217→   217→    return { subject, enrolled, mark, grade, ca: comp?.ca ?? null, exam: comp?.exam ?? null };
   218→   218→  });
   219→   219→  // A subject with no mark at all (both components ABS/NA/NR/PENDING) is
   220→   220→  // omitted from the card entirely; non-enrolled subsidiaries stay with "—".
   221→   221→  // Remaining rows renumber continuously after any omitted subject (the #
   222→   222→  // column below).
   223→   223→  const rows = allRows.filter((r) => !r.enrolled || r.mark !== null);
   224→   224→  const assessed = allRows.filter((r) => r.mark !== null);
   225→   225→  const pointsTotal = assessed.reduce((a, r) => a + (r.grade?.points ?? 0), 0);
   226→   226→  // Secondary division: best 7 subject points; null when fewer than 7 graded
   227→   227→  // (and always null for primary classes — divisions are a NECTA outcome).
   228→   228→  const division = isPrimary ? null : divisionFor(assessed.map((r) => r.grade?.points ?? null));
   229→   229→  const passed = subjectsPassed(assessed.map((r) => r.grade?.grade ?? null));
   230→   230→  // Strongest subject: highest total; ties go to the first in subject order.
   231→   231→  const strongest =
   232→   232→    assessed.length > 0
   233→   233→      ? assessed.reduce((best, r) => ((r.mark as number) > (best.mark as number) ? r : best))
   234→   234→      : undefined;
   235→   235→  // Points and division are hidden on primary cards (Primary A–E bands carry
   236→   236→  // no points) and on any secondary class whose teacher set the Report
   237→   237→  // Overview "No Division" flag to Yes.
   238→   238→  const noDivision = !isPrimary && (classById(student.classId)?.noDivision ?? false);
   239→   239→  const showPoints = !isPrimary && !noDivision;
   240→   240→  // Most improved (primary only): largest CA delta C2 − C1 per subject; a
   241→   241→  // single recorded term counts as 0. Ties go to the first subject in order.
   242→   242→  const mostImproved = (() => {
   243→   243→    if (!isPrimary) return null;
   244→   244→    const deltas = classSubjects.map((cs) => {
   245→   245→      const comp = getComponents(student.classId, cs.subjectId)[student.id];
   246→   246→      const cur = comp?.enrolled ? comp.ca ?? null : null;
   247→   247→      const prev = prevCaOf(student.classId, cs.subjectId)[student.id] ?? null;
   248→   248→      return { subjectId: cs.subjectId, delta: cur !== null && prev !== null ? cur - prev : 0 };
   249→   249→    });
   250→   250→    const best = mostImprovedSubject(deltas);
   251→   251→    return best && best.delta > 0 ? best : null;
   252→   252→  })();
   253→   253→  const mostImprovedSubjectRow = mostImproved
   254→   254→    ? rows.find((r) => r.subject.id === mostImproved.subjectId)
   255→   255→    : undefined;
   256→   256→
   257→   257→  // Chart data (A4 section 6 / chart prompt §3–§4): achieved total vs class
   258→   258→  // average and class top per assessed subject, over the student's enrolled
   259→   259→  // classmates. Class stats follow the cycle's engine: mid-term counts a
   260→   260→  // student with a single assessment at 100%; terminal only counts students
   261→   261→  // with BOTH components (chart prompt §4/§9).
   262→   262→  const variant: ChartVariant = cycle.kind === "mid-term" ? "mid-term" : "terminal";
   263→   263→  const chartRows: ChartRow[] = assessed.map((r) => {
   264→   264→    const cs = classSubjects.find((c) => c.subjectId === r.subject.id);
   265→   265→    const comps = getComponents(student.classId, r.subject.id);
   266→   266→    const scores = Object.values(comps)
   267→   267→      .filter((c) => c.enrolled)
   268→   268→      .map((c) => classScoreFor(variant, c.ca, c.exam, cs?.caMarks ?? 25, cs?.examMarks ?? 75))
   269→   269→      .filter((t): t is number => t !== null);
   270→   270→    return {
   271→   271→      name: r.subject.name,
   272→   272→      achieved: r.mark ?? 0,
   273→   273→      classAvg: scores.length ? scores.reduce((a, b) => a + b, 0) / scores.length : 0,
   274→   274→      classTop: scores.length ? Math.max(...scores) : 0,
   275→   275→    };
   276→   276→  });
   277→   277→  // Series + static chart image (chart prompt §7 call sequence).
   278→   278→  const chartSeries = buildChartSeries(
   279→   279→    chartRows.map((r) => ({ name: r.name, achieved: r.achieved })),
   280→   280→    chartRows.map((r) => r.classAvg),
   281→   281→    chartRows.map((r) => r.classTop),
   282→   282→    strongest?.subject.name ?? "",
   283→   283→    variant,
   284→   284→  );
   285→   285→  const chartImage = generateChartImage(chartSeries, variant);
   286→   286→  // Best-Improvement badge: Mid-Term primary cards only (chart prompt §8).
   287→   287→  const chartBestImprovement =
   288→   288→    variant === "mid-term" && mostImproved && mostImprovedSubjectRow
   289→   289→      ? { name: mostImprovedSubjectRow.subject.name, value: mostImproved.delta }
   290→   290→      : null;
   291→   291→
   292→   292→  const switchStudent = (id: string) => {
   293→   293→    router.push(`/academic/report-cycles/${cycle.id}/students/${id}/report-card`);
   294→   294→  };
   295→   295→
   296→   296→  const downloadPdf = usePrintDownload(
   297→   297→    "Use 'Save as PDF' in the print dialog to download the report card.",
   298→   298→  );
   299→   299→
   300→   300→  // Performance summary cells; the points/division pair appears only where
   301→   301→  // points are printed (secondary classes without the No-Division flag).
   302→   302→  const summaryCells = [
   303→   303→    { label: "Total marks", value: result ? fmtTotalMark(result.total) : "-", accent: false },
   304→   304→    { label: "Average", value: result ? result.average.toFixed(1) : "-", accent: false },
   305→   305→    {
   306→   306→      label: "Overall grade",
   307→   307→      value: result && result.count > 0 ? result.grade : "-",
   308→   308→      accent: true,
   309→   309→    },
   310→   310→    ...(showPoints
   311→   311→      ? [
   312→   312→          {
   313→   313→            label: "Points total",
   314→   314→            value: assessed.length > 0 ? String(pointsTotal) : "-",
   315→   315→            accent: false,
   316→   316→          },
   317→   317→          {
   318→   318→            label: "Division",
   319→   319→            value: division ? `Div ${division.division}` : "—",
   320→   320→            accent: true,
   321→   321→          },
   322→   322→        ]
   323→   323→      : []),
   324→   324→    {
   325→   325→      label: "Subjects passed",
   326→   326→      value: assessed.length > 0 ? `${passed} of ${assessed.length}` : "-",
   327→   327→      accent: false,
   328→   328→    },
   329→   329→    { label: "Class size", value: String(classSize), accent: false },
   330→   330→  ];
   331→   331→
   332→   332→  // Attendance row cells; "Not recorded" appears only when > 0 (A3.8).
   333→   333→  const rateText = attendance.rate === null ? "–" : `${attendance.rate.toFixed(1)}%`;
   334→   334→  const attRatingText =
   335→   335→    attendance.rating === "none" ? null : ATTENDANCE_RATING_TEXT[attendance.rating];
   336→   336→  const attCells = [
   337→   337→    {
   338→   338→      label: "Days attended / recorded",
   339→   339→      value: `${attendance.daysAttended} / ${attendance.daysRecorded}`,
   340→   340→    },
   341→   341→    { label: "Attendance rate", value: rateText },
   342→   342→    { label: "Late", value: String(attendance.late) },
   343→   343→    { label: "Absent", value: String(attendance.absent) },
   344→   344→    { label: "Excused", value: String(attendance.excused) },
   345→   345→    ...(attendance.notRecorded > 0
   346→   346→      ? [
   347→   347→          {
   348→   348→            label: "Not recorded",
   349→   349→            value: `${attendance.notRecorded} day${attendance.notRecorded === 1 ? "" : "s"}`,
   350→   350→          },
   351→   351→        ]
   352→   352→      : []),
   353→   353→  ];
   354→   354→
   355→   355→  return (
   356→   356→    <div className="space-y-5">
   357→   357→      <PageHeader
   358→   358→        title="Report card"
   359→   359→        description={`${student.name} · ${classNameOf(student.classId)} · ${cycle.name}`}
   360→   360→      />
   361→   361→
   362→   362→      {/* Toolbar: screen only, never printed */}
   363→   363→      <div className="flex flex-col gap-3 rounded-lg border border-border/80 bg-white p-3 sm:flex-row sm:items-center sm:justify-between print:hidden">
   364→   364→        <div className="flex flex-wrap items-center gap-2">
   365→   365→          <span className="text-xs font-medium text-muted-foreground">Student</span>
   366→   366→          {/* ds-allow-native-select: shadcn Select is a Radix listbox trigger, not a native select */}
   367→   367→          <Select value={student.id} onValueChange={switchStudent}>
   368→   368→            <SelectTrigger className="h-8 w-full text-compact sm:w-65">
   369→   369→              <SelectValue />
   370→   370→            </SelectTrigger>
   371→   371→            <SelectContent>
   372→   372→              {classmates.map((s) => (
   373→   373→                <SelectItem key={s.id} value={s.id}>
   374→   374→                  {s.name}
   375→   375→                </SelectItem>
   376→   376→              ))}
   377→   377→            </SelectContent>
   378→   378→          </Select>
   379→   379→        </div>
   380→   380→        <div className="flex items-center gap-2">
   381→   381→          <Button size="sm" className="gap-1.5" onClick={downloadPdf}>
   382→   382→            <Download className="size-4" /> Download PDF
   383→   383→          </Button>
   384→   384→          <Button size="sm" variant="outline" className="gap-1.5" onClick={() => window.print()}>
   385→   385→            <Printer className="size-4" /> Print
   386→   386→          </Button>
   387→   387→        </div>
   388→   388→      </div>
   389→   389→
   390→   390→      {/* Legacy rule 7: a student with no marks at all renders an empty

   392→   392→      {assessed.length === 0 ? (
   393→   393→        <div className="flex items-center gap-2 rounded-lg border border-destructive/40 bg-destructive/8 px-3 py-2 text-compact text-destructive print:hidden">
   394→   394→          <AlertTriangle className="size-4 shrink-0" />
   395→   395→          No marks have been entered for this student yet — report cards are generated only after marks exist.
   396→   396→        </div>
   397→   397→      ) : null}
   398→   398→
   399→   399→      {/* The printable card: scrollable on a phone, exact A4 sheet in print.
   400→   400→          Totally white paper; every section is a bordered table. */}
   401→   401→      <div className="overflow-x-auto print:overflow-visible">
   402→   402→        <div className="print-area mx-auto w-full max-w-205 border border-border bg-white p-8 font-serif text-foreground shadow-sm">
   403→   403→          {/* School header: logo + school block on white */}
   404→   404→          <div className="flex items-center gap-4">
   405→   405→            <img
   406→   406→              src="/josephian-logo.png"
   407→   407→              alt="Josephian Schools logo"
   408→   408→              className="h-20 w-20 shrink-0"
   409→   409→            />
   410→   410→            <div className="min-w-0 flex-1 text-center">
