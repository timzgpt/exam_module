    28→function hashStr(s: string): number {
    29→  let h = 2166136261;
    30→  for (let i = 0; i < s.length; i++) {
    31→    h ^= s.charCodeAt(i);
    32→    h = Math.imul(h, 16777619);
    33→  }
    34→  return h >>> 0;
    35→}
    36→
    37→function mulberry32(seed: number): () => number {
    38→  let a = seed;
    39→  return () => {
    40→    a |= 0;
    41→    a = (a + 0x6d2b79f5) | 0;
    42→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    43→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    44→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    45→  };
    46→}

   315→
   316→/** Subjects that are subsidiary (optional) for every class in the demo. */
   317→const SUBSIDIARY_SUBJECT_IDS = new Set(["s9"]);
   318→
   319→/** Students forced onto a subsidiary subject regardless of the seeded draw. */
   320→const ENROL_OVERRIDES = new Set(["st1|s9", "f1a-04|s9"]);
   321→
   322→/** Pairs with missing entries so deadline tracker / readiness look real. */
   323→const NULL_PATCHES: Record<string, string[] | "ALL"> = {
   324→  "cls-4a|s5": ["f4a-03", "f4a-07", "f4a-10", "f4a-11"],
   325→  "cls-2a|s7": "ALL",
   326→};
   327→
   328→/** DISQ malpractice: a real zero that IS included in averages (0.0). */
   329→const DISQ_PATCHES = new Set(["cls-1a|s2|f1a-04"]);
   330→
   331→/**
   332→ * Is this student enrolled in the class subject? Core subjects are compulsory
   333→ * (every student takes them); subsidiaries are opt-in per student, seeded so
   334→ * every page sees the same enrolment.
   335→ */
   336→export function isEnrolledIn(
   337→  classId: string,
   338→  subjectId: string,
   339→  studentId: string,
   340→): boolean {
   341→  if (!SUBSIDIARY_SUBJECT_IDS.has(subjectId)) return true;
   342→  if (ENROL_OVERRIDES.has(`${studentId}|${subjectId}`)) return true;
   343→  return hashStr(`enrol|${classId}|${subjectId}|${studentId}`) % 10 < 6;
   344→}

   346→const marksCache = new Map<string, Record<string, number | null>>();
   347→
   348→export function getMarks(
   349→  classId: string,
   350→  subjectId: string,
   351→): Record<string, number | null> {
   352→  const key = `${classId}|${subjectId}`;
   353→  const cached = marksCache.get(key);
   354→  if (cached) return cached;
   355→
   356→  const rnd = mulberry32(hashStr(key));
   357→  const classOffset = (hashStr(classId) % 13) - 6;
   358→  const subjectOffset = (hashStr(subjectId) % 15) - 7;
   359→  const result: Record<string, number | null> = {};
   360→  for (const s of studentsInClass(classId)) {
   361→    if (!isEnrolledIn(classId, subjectId, s.id)) {
   362→      result[s.id] = null; // not taking a subsidiary subject — dash everywhere
   363→      continue;
   364→    }
   365→    const ability = 44 + (hashStr(s.id) % 46); // 44-89 latent ability
   366→    const jitter = rnd() * 18 - 9;
   367→    const ca = Math.max(0, Math.min(100, Math.round(ability + classOffset + jitter)));
   368→    const exam = Math.max(0, Math.min(100, ca + Math.round(rnd() * 16 - 8)));
   369→    // Subject total via the 30/70 weighting so components round-trip exactly.
   370→    result[s.id] = subjectTotal(ca, exam, 30, 70);
   371→  }
   372→  const patch = NULL_PATCHES[key];
   373→  if (patch === "ALL") {
   374→    for (const sid of Object.keys(result)) result[sid] = null;
   375→  } else if (patch) {
   376→    for (const sid of patch) result[sid] = null;
   377→  }
   378→  for (const sid of Object.keys(result)) {
   379→    if (DISQ_PATCHES.has(`${key}|${sid}`)) result[sid] = 0; // DISQ is a real zero
   380→  }
   381→  marksCache.set(key, result);
   382→  return result;
   383→}

   389→/** Per-student status codes shown in the marks grid instead of a blank. */
   390→const STATUS_PATCHES: Record<string, Record<string, MarkStatusCode>> = {

   394→/**
   395→ * Status code typed in place of a mark. ABS/NA/NR/PENDING are treated as
   396→ * blank (null); DISQ is a real zero. Non-enrolled subsidiary students show
   397→ * "—" and never carry a status.
   398→ */
   399→export function markStatusOf(
   400→  classId: string,
   401→  subjectId: string,
   402→  studentId: string,
   403→): MarkStatusCode | null {
   404→  if (`${classId}|${subjectId}` === "cls-2a|s7") return "PENDING";
   405→  if (DISQ_PATCHES.has(`${classId}|${subjectId}|${studentId}`)) return "DISQ";
   406→  return STATUS_PATCHES[`${classId}|${subjectId}`]?.[studentId] ?? null;
   407→}

   409→export interface MarkComponents {
   410→  enrolled: boolean;
   411→  /** Continuous assessment, entered out of 100 (weighted 30%). */
   412→  ca: number | null;
   413→  /** End-of-term written exam, entered out of 100 (weighted 70%). */
   414→  exam: number | null;
   415→  status: MarkStatusCode | null;
   416→}
   417→
   418→const componentsCache = new Map<string, Record<string, MarkComponents>>();
   419→
   420→/**
   421→ * CA / Exam components per student. Components round-trip exactly to the
   422→ * stored subject total via subjectTotal(ca, exam, 30, 70).
   423→ */
   424→export function getComponents(
   425→  classId: string,
   426→  subjectId: string,
   427→): Record<string, MarkComponents> {
   428→  const key = `${classId}|${subjectId}`;
   429→  const cached = componentsCache.get(key);
   430→  if (cached) return cached;
   431→  const marks = getMarks(classId, subjectId);
   432→  const result: Record<string, MarkComponents> = {};
   433→  for (const [studentId, total] of Object.entries(marks)) {
   434→    const enrolled = isEnrolledIn(classId, subjectId, studentId);
   435→    const status = markStatusOf(classId, subjectId, studentId);
   436→    if (total === null || !enrolled) {
   437→      result[studentId] = { enrolled, ca: null, exam: null, status };
   438→      continue;
   439→    }
   440→    // Invert the weighting exactly: search integer CA/Exam pairs near the
   441→    // total until round(CA x 0.3 + Exam x 0.7, 2) reproduces the total.
   442→    let ca = Math.max(0, Math.min(100, Math.round(total)));
   443→    let exam = Math.max(0, Math.min(100, Math.round((total - 0.3 * ca) / 0.7)));
   444→    for (let c = Math.max(0, Math.round(total) - 10); c <= Math.min(100, Math.round(total) + 10); c++) {
   445→      const e = Math.round((total - 0.3 * c) / 0.7);
   446→      if (e < 0 || e > 100) continue;
   447→      if (Math.round((0.3 * c + 0.7 * e) * 100) / 100 === total) {
   448→        ca = c;
   449→        exam = e;
   450→        break;
   451→      }
   452→    }
   453→    result[studentId] = { enrolled, ca, exam, status };
   454→  }
   455→  componentsCache.set(key, result);
   456→  return result;
   457→}

   809→export interface StudentResult {
   810→  student: Student;
   811→  marks: Record<string, number | null>;
   812→  total: number;
   813→  count: number;
   814→  average: number;
   815→  grade: string;
   816→  rank: number;
   817→  /** Sum of the best 7 subject points; null when fewer than 7 graded. */
   818→  bestSevenPoints: number | null;
   819→  /** NECTA division from the best 7 subject points; null when < 7 graded. */
   820→  division: DivisionResult | null;
   821→  /** Subjects whose grade letter is not E or F. */
   822→  passedCount: number;
   823→}
   824→
   825→const classResultsCache = new Map<string, StudentResult[]>();
   826→
   827→export function classResults(classId: string): StudentResult[] {
   828→  const cached = classResultsCache.get(classId);
   829→  if (cached) return cached;
   830→  const rows = studentsInClass(classId).map((student) => {
   831→    const marks: Record<string, number | null> = {};
   832→    const points: (number | null)[] = [];
   833→    const grades: (string | null)[] = [];
   834→    let total = 0;
   835→    let count = 0;
   836→    for (const sub of subjects) {
   837→      const m = getMarks(classId, sub.id)[student.id];
   838→      marks[sub.id] = m;
   839→      if (m !== null) {
   840→        total += m;
   841→        count += 1;
   842→        const g = gradeFor(m, defaultScheme);
   843→        grades.push(g?.grade ?? null);
   844→        points.push(g?.points ?? null);
   845→      } else {
   846→        grades.push(null);
   847→        points.push(null);
   848→      }
   849→    }
   850→    const average = count > 0 ? total / count : 0;
   851→    const division = divisionFor(points);
   852→    return {
   853→      student,
   854→      marks,
   855→      total,
   856→      count,
   857→      average,
   858→      grade: count > 0 ? gradeFor(average, defaultScheme)?.grade ?? "F" : "-",
   859→      rank: 0,
   860→      bestSevenPoints: division?.pointsSum ?? null,
   861→      division,
   862→      passedCount: subjectsPassed(grades),
   863→    };
   864→  });
   865→  rows.sort((a, b) => b.average - a.average);
   866→  rows.forEach((r, i) => (r.rank = i + 1));
   867→  classResultsCache.set(classId, rows);
   868→  return rows;
   869→}

   907→export interface CycleStats {

   910→  passRate: number;
   911→  gradeDistribution: Record<string, number>;
   912→  classAverages: { classId: string; name: string; avg: number }[];

   919→const cycleStatsCache = new Map<string, CycleStats>();
   920→
   921→export function cycleStatsFor(cycleId: string): CycleStats {
   922→  const cached = cycleStatsCache.get(cycleId);
   923→  if (cached) return cached;
   924→  const all = classes.flatMap((c) => classResults(c.id));
   925→  const allMarks = all.flatMap((r) => Object.values(r.marks).filter((v): v is number => v !== null));
   926→  const average = allMarks.length ? allMarks.reduce((a, b) => a + b, 0) / allMarks.length : 0;
   927→  const gradeDistribution: Record<string, number> = {};
   928→  for (const b of boundaries) gradeDistribution[b.grade] = 0;
   929→  for (const m of allMarks) {
   930→    const g = gradeFor(m, defaultScheme)?.grade;
   931→    if (g) gradeDistribution[g] += 1;
   932→  }
   933→  const totalEntries = marksDeadlines.reduce((a, d) => a + d.entriesTotal, 0);
   934→  const doneEntries = marksDeadlines.reduce((a, d) => a + d.entriesDone, 0);
   935→  const commentTotals = classes.reduce(
   936→    (acc, c) => {
   937→      const p = commentProgress(c.id);
   938→      return { done: acc.done + p.done, total: acc.total + p.total };
   939→    },
   940→    { done: 0, total: 0 },
   941→  );
   942→  const stats: CycleStats = {
   943→    totalStudents: all.length,
   944→    average,
   945→    passRate: allMarks.length ? Math.round((allMarks.filter((m) => m >= defaultScheme.passMark).length / allMarks.length) * 100) : 0,
   946→    gradeDistribution,
   947→    classAverages: classes.map((c) => ({
   948→      classId: c.id,
   949→      name: c.name,
   950→      avg: classResults(c.id).reduce((a, r) => a + r.average, 0) / classResults(c.id).length,
   951→    })),
   952→    subjectAverages: subjects.map((sub) => {
   953→      const ms = classes.flatMap((c) =>
   954→        Object.values(getMarks(c.id, sub.id)).filter((v): v is number => v !== null),
   955→      );
   956→      return {
   957→        subjectId: sub.id,
   958→        name: sub.name,
   959→        department: sub.department,
   960→        avg: ms.length ? ms.reduce((a, b) => a + b, 0) / ms.length : 0,
   961→        passRate: ms.length ? Math.round((ms.filter((m) => m >= defaultScheme.passMark).length / ms.length) * 100) : 0,
   962→      };
   963→    }),
   964→    marksCompletion: totalEntries ? Math.round((doneEntries / totalEntries) * 100) : 0,
   965→    commentsCompletion: commentTotals.total ? Math.round((commentTotals.done / commentTotals.total) * 100) : 0,
   966→    topStudents: all
   967→      .slice()
   968→      .sort((a, b) => b.average - a.average)
   969→      .slice(0, 5)
   970→      .map((r) => ({ name: r.student.name, className: classNameOf(r.student.classId), avg: r.average })),
   971→  };
   972→  cycleStatsCache.set(cycleId, stats);
   973→  return stats;
   974→}

  1000→    });
