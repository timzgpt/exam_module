    31→function hashStr(s: string): number {
    32→  let h = 2166136261;
    33→  for (let i = 0; i < s.length; i++) {
    34→    h ^= s.charCodeAt(i);
    35→    h = Math.imul(h, 16777619);
    36→  }
    37→  return h >>> 0;
    38→}
    39→
    40→function mulberry32(seed: number): () => number {
    41→  let a = seed;
    42→  return () => {
    43→    a |= 0;
    44→    a = (a + 0x6d2b79f5) | 0;
    45→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    46→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    47→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    48→  };
    49→}

   356→/** Subjects that are subsidiary (optional) for every class in the demo. */
   357→const SUBSIDIARY_SUBJECT_IDS = new Set(["s9"]);

   359→/** Students forced onto a subsidiary subject regardless of the seeded draw. */
   360→const ENROL_OVERRIDES = new Set(["st1|s9", "f1a-04|s9"]);
   361→
   362→/** Pairs with missing entries so deadline tracker / readiness look real. */
   363→const NULL_PATCHES: Record<string, string[] | "ALL"> = {
   364→  "cls-4a|s5": ["f4a-03", "f4a-07", "f4a-10", "f4a-11"],
   365→  "cls-2a|s7": "ALL",
   366→};

   368→/** DISQ malpractice: a real zero that IS included in averages (0.0). */
   369→const DISQ_PATCHES = new Set(["cls-1a|s2|f1a-04"]);

   371→/**
   372→ * Is this student enrolled in the class subject? Core subjects are compulsory
   373→ * (every student takes them); subsidiaries are opt-in per student, seeded so
   374→ * every page sees the same enrolment.
   375→ */
   376→export function isEnrolledIn(
   377→  classId: string,
   378→  subjectId: string,
   379→  studentId: string,
   380→): boolean {
   381→  if (!SUBSIDIARY_SUBJECT_IDS.has(subjectId)) return true;
   382→  if (ENROL_OVERRIDES.has(`${studentId}|${subjectId}`)) return true;
   383→  return hashStr(`enrol|${classId}|${subjectId}|${studentId}`) % 10 < 6;
   384→}
   385→
   386→const marksCache = new Map<string, Record<string, number | null>>();
   387→
   388→export function getMarks(
   389→  classId: string,
   390→  subjectId: string,
   391→): Record<string, number | null> {
   392→  const key = `${classId}|${subjectId}`;
   393→  const cached = marksCache.get(key);
   394→  if (cached) return cached;
   395→
   396→  const rnd = mulberry32(hashStr(key));
   397→  const classOffset = (hashStr(classId) % 13) - 6;
   398→  const subjectOffset = (hashStr(subjectId) % 15) - 7;
   399→  const result: Record<string, number | null> = {};
   400→  for (const s of studentsInClass(classId)) {
   401→    if (!isEnrolledIn(classId, subjectId, s.id)) {
   402→      result[s.id] = null; // not taking a subsidiary subject — dash everywhere
   403→      continue;
   404→    }
   405→    const ability = 44 + (hashStr(s.id) % 46); // 44-89 latent ability
   406→    const jitter = rnd() * 18 - 9;
   407→    const ca = Math.max(0, Math.min(100, Math.round(ability + classOffset + jitter)));
   408→    const exam = Math.max(0, Math.min(100, ca + Math.round(rnd() * 16 - 8)));
   409→    // Subject total via the class subject's own 25/75 split so components
   410→    // round-trip exactly (Monthly CA 25% + Exam 75% for every demo subject).
   411→    const cs = classSubjectsFor(classId).find((c) => c.subjectId === subjectId);
   412→    result[s.id] = subjectTotal(ca, exam, cs?.caMarks ?? 25, cs?.examMarks ?? 75);
   413→  }
   414→  const patch = NULL_PATCHES[key];
   415→  if (patch === "ALL") {
   416→    for (const sid of Object.keys(result)) result[sid] = null;
   417→  } else if (patch) {
   418→    for (const sid of patch) result[sid] = null;
   419→  }
   420→  for (const sid of Object.keys(result)) {
   421→    if (DISQ_PATCHES.has(`${key}|${sid}`)) result[sid] = 0; // DISQ is a real zero
   422→  }
   423→  marksCache.set(key, result);
   424→  return result;
   425→}

   431→/** Per-student status codes shown in the marks grid instead of a blank. */
   432→const STATUS_PATCHES: Record<string, Record<string, MarkStatusCode>> = {
   433→  "cls-4a|s5": { "f4a-03": "ABS", "f4a-07": "ABS", "f4a-10": "NA", "f4a-11": "NA" },

   436→/**
   437→ * Status code typed in place of a mark. ABS/NA/NR/PENDING are treated as
   438→ * blank (null); DISQ is a real zero. Non-enrolled subsidiary students show
   439→ * "—" and never carry a status.

   441→export function markStatusOf(
   442→  classId: string,
   443→  subjectId: string,
   444→  studentId: string,
   445→): MarkStatusCode | null {
   446→  if (`${classId}|${subjectId}` === "cls-2a|s7") return "PENDING";
   447→  if (DISQ_PATCHES.has(`${classId}|${subjectId}|${studentId}`)) return "DISQ";
   448→  return STATUS_PATCHES[`${classId}|${subjectId}`]?.[studentId] ?? null;
   449→}

   451→export interface MarkComponents {
   452→  enrolled: boolean;
   453→  /** Monthly / continuous assessment, entered out of 100 (weighted 25%). */
   454→  ca: number | null;

   460→const componentsCache = new Map<string, Record<string, MarkComponents>>();
   461→
   462→/**
   463→ * CA / Exam components per student. Components round-trip exactly to the
   464→ * stored subject total via subjectTotal(ca, exam, w1, w2) using the class
   465→ * subject's own 25/75 assessment split.
   466→ */
   467→export function getComponents(
   468→  classId: string,
   469→  subjectId: string,
   470→): Record<string, MarkComponents> {
   471→  const key = `${classId}|${subjectId}`;
   472→  const cached = componentsCache.get(key);
   473→  if (cached) return cached;
   474→  const marks = getMarks(classId, subjectId);
   475→  const cs = classSubjectsFor(classId).find((c) => c.subjectId === subjectId);
   476→  const w1 = (cs?.caMarks ?? 25) / 100;
   477→  const w2 = (cs?.examMarks ?? 75) / 100;
   478→  const result: Record<string, MarkComponents> = {};
   479→  for (const [studentId, total] of Object.entries(marks)) {
   480→    const enrolled = isEnrolledIn(classId, subjectId, studentId);
   481→    const status = markStatusOf(classId, subjectId, studentId);
   482→    if (total === null || !enrolled) {
   483→      result[studentId] = { enrolled, ca: null, exam: null, status };
   484→      continue;
   485→    }
   486→    // Invert the weighting exactly: search integer CA/Exam pairs near the
   487→    // total until round(CA x w1 + Exam x w2, 2) reproduces the total.
   488→    let ca = Math.max(0, Math.min(100, Math.round(total)));
   489→    let exam = Math.max(0, Math.min(100, Math.round((total - w1 * ca) / w2)));
   490→    for (let c = Math.max(0, Math.round(total) - 10); c <= Math.min(100, Math.round(total) + 10); c++) {
   491→      const e = Math.round((total - w1 * c) / w2);
   492→      if (e < 0 || e > 100) continue;
   493→      if (Math.round((w1 * c + w2 * e) * 100) / 100 === total) {
   494→        ca = c;
   495→        exam = e;
   496→        break;
   497→      }
   498→    }
   499→    result[studentId] = { enrolled, ca, exam, status };
   500→  }
   501→  componentsCache.set(key, result);
   502→  return result;
   503→}

   520→/**

   523→ * this term's CA so the delta lands in a plausible −10…+18 band. Students
   524→ * without a previous record (new arrivals) get null — their delta counts as
   525→ * 0, never as an improvement.

   527→export function prevCaOf(
   528→  classId: string,
   529→  subjectId: string,
   530→): Record<string, number | null> {
   531→  const key = `prevca|${classId}|${subjectId}`;
   532→  const cached = prevCaCache.get(key);
   533→  if (cached) return cached;
   534→  const comps = getComponents(classId, subjectId);
   535→  const result: Record<string, number | null> = {};
   536→  for (const [studentId, comp] of Object.entries(comps)) {
   537→    if (!comp.enrolled || comp.ca === null) {
   538→      result[studentId] = null;
   539→      continue;
   540→    }
   541→    const delta = (hashStr(`prev|${key}|${studentId}`) % 29) - 10; // −10…+18
   542→    result[studentId] = Math.max(0, Math.min(100, comp.ca - delta));
   543→  }
   544→  prevCaCache.set(key, result);
   545→  return result;

   890→export interface StudentResult {

   899→  bestSevenPoints: number | null;
   900→  /** NECTA division from the best 7 subject points; null when < 7 graded. */

   908→export function classResults(classId: string): StudentResult[] {
   909→  const cached = classResultsCache.get(classId);
   910→  if (cached) return cached;
   911→  const scheme = schemeForClass(classId);
   912→  // Divisions are a secondary (NECTA CSEE) outcome — never computed for primary.
   913→  const allowDivision = !isPrimaryClass(classId);
   914→  const rows = studentsInClass(classId).map((student) => {
   915→    const marks: Record<string, number | null> = {};
   916→    const points: (number | null)[] = [];
   917→    const grades: (string | null)[] = [];
   918→    let total = 0;
   919→    let count = 0;
   920→    for (const sub of subjects) {
   921→      const m = getMarks(classId, sub.id)[student.id];
   922→      marks[sub.id] = m;
   923→      if (m !== null) {
   924→        total += m;
   925→        count += 1;
   926→        const g = gradeFor(m, scheme);
   927→        grades.push(g?.grade ?? null);
   928→        points.push(g?.points ?? null);
   929→      } else {
   930→        grades.push(null);
   931→        points.push(null);
   932→      }
   933→    }
   934→    const average = count > 0 ? total / count : 0;
   935→    const division = allowDivision ? divisionFor(points) : null;
   936→    return {
   937→      student,
   938→      marks,
   939→      total,
   940→      count,
   941→      average,
   942→      grade: count > 0 ? gradeFor(average, scheme)?.grade ?? scheme.boundaries[scheme.boundaries.length - 1].grade : "-",
   943→      rank: 0,
   944→      bestSevenPoints: division?.pointsSum ?? null,
   945→      division,
   946→      passedCount: subjectsPassed(grades),
   947→    };
   948→  });
   949→  rows.sort((a, b) => b.average - a.average);
   950→  rows.forEach((r, i) => (r.rank = i + 1));
   951→  classResultsCache.set(classId, rows);
   952→  return rows;
   953→}

   961→/**
   962→ * Per-subject class statistics, in CLASS-SUBJECT order (the order set on the
   963→ * Class subjects page — not the catalogue order) and graded with the class's
   964→ * own scheme. Marks not entered are doubly excluded from every average.
   965→ */
   966→export function subjectAveragesFor(classId: string): {
   967→  subjectId: string;
   968→  avg: number;
   969→  passRate: number;
   970→  highest: number;
   971→  lowest: number;
   972→  distribution: Record<string, number>;
   973→}[] {
   974→  const scheme = schemeForClass(classId);
   975→  return classSubjectsFor(classId).map((cs) => {
   976→    const marks = Object.values(getMarks(classId, cs.subjectId)).filter(
   977→      (v): v is number => v !== null,
   978→    );
   979→    const avg = marks.length ? marks.reduce((a, b) => a + b, 0) / marks.length : 0;
   980→    const distribution: Record<string, number> = {};
   981→    for (const b of scheme.boundaries) distribution[b.grade] = 0;
   982→    for (const m of marks) {
   983→      const g = gradeFor(m, scheme)?.grade;
   984→      if (g) distribution[g] += 1;
   985→    }
   986→    return {
   987→      subjectId: cs.subjectId,
   988→      avg,
   989→      passRate: marks.length ? Math.round((marks.filter((m) => m >= scheme.passMark).length / marks.length) * 100) : 0,
   990→      highest: marks.length ? Math.max(...marks) : 0,
   991→      lowest: marks.length ? Math.min(...marks) : 0,
   992→      distribution,
   993→    };
   994→  });
   995→}

  1011→export function cycleStatsFor(cycleId: string): CycleStats {
  1012→  const cached = cycleStatsCache.get(cycleId);
  1013→  if (cached) return cached;
  1014→  const all = classes.flatMap((c) => classResults(c.id));
  1015→  const allMarks = all.flatMap((r) => Object.values(r.marks).filter((v): v is number => v !== null));
  1016→  const average = allMarks.length ? allMarks.reduce((a, b) => a + b, 0) / allMarks.length : 0;
  1017→  // Grade each class's marks with THAT class's scheme (primary A-E vs NECTA
  1018→  // CSEE A-F) so the school-wide distribution never mis-files a primary mark.
  1019→  const gradeDistribution: Record<string, number> = {};
  1020→  for (const s of gradingSchemes) for (const b of s.boundaries) gradeDistribution[b.grade] = 0;
  1021→  let passedEntries = 0;
  1022→  for (const c of classes) {
  1023→    const scheme = schemeForClass(c.id);
  1024→    for (const r of classResults(c.id)) {
  1025→      for (const m of Object.values(r.marks)) {
  1026→        if (m === null) continue;
  1027→        const g = gradeFor(m, scheme)?.grade;
  1028→        if (g) gradeDistribution[g] = (gradeDistribution[g] ?? 0) + 1;
  1029→        if (m >= scheme.passMark) passedEntries += 1;
  1030→      }
  1031→    }
  1032→  }
  1033→  const totalEntries = marksDeadlines.reduce((a, d) => a + d.entriesTotal, 0);
  1034→  const doneEntries = marksDeadlines.reduce((a, d) => a + d.entriesDone, 0);
  1035→  const commentTotals = classes.reduce(
  1036→    (acc, c) => {
  1037→      const p = commentProgress(c.id);
  1038→      return { done: acc.done + p.done, total: acc.total + p.total };
  1039→    },
  1040→    { done: 0, total: 0 },
  1041→  );
  1042→  const stats: CycleStats = {
  1043→    totalStudents: all.length,
  1044→    average,
  1045→    passRate: allMarks.length ? Math.round((passedEntries / allMarks.length) * 100) : 0,
  1046→    gradeDistribution,
  1047→    classAverages: classes.map((c) => ({
  1048→      classId: c.id,
  1049→      name: c.name,
  1050→      avg: classResults(c.id).reduce((a, r) => a + r.average, 0) / classResults(c.id).length,
  1051→    })),
  1052→    subjectAverages: subjects.map((sub) => {
  1053→      let entered = 0;
  1054→      let passed = 0;
  1055→      let sum = 0;
  1056→      for (const c of classes) {
  1057→        const scheme = schemeForClass(c.id);
  1058→        for (const m of Object.values(getMarks(c.id, sub.id))) {
  1059→          if (m === null) continue;
  1060→          entered += 1;
  1061→          sum += m;
  1062→          if (m >= scheme.passMark) passed += 1;
  1063→        }
  1064→      }
  1065→      return {
  1066→        subjectId: sub.id,
  1067→        name: sub.name,
  1068→        department: sub.department,
  1069→        avg: entered ? sum / entered : 0,
  1070→        passRate: entered ? Math.round((passed / entered) * 100) : 0,
  1071→      };
  1072→    }),
  1073→    marksCompletion: totalEntries ? Math.round((doneEntries / totalEntries) * 100) : 0,
  1074→    commentsCompletion: commentTotals.total ? Math.round((commentTotals.done / commentTotals.total) * 100) : 0,
  1075→    topStudents: all
  1076→      .slice()
  1077→      .sort((a, b) => b.average - a.average)
  1078→      .slice(0, 5)
  1079→      .map((r) => ({ name: r.student.name, className: classNameOf(r.student.classId), avg: r.average })),
  1080→  };
  1081→  cycleStatsCache.set(cycleId, stats);
  1082→  return stats;
  1083→}

  1161→/**
  1162→ * Per-student attendance summary for the current demo cycle, aggregated from
  1163→ * a deterministic fake daily roll call over the cycle's attendance period
  1164→ * (A3.8): counted marks only; late counts as attended; excused stays in the
  1165→ * denominator; possible days with no counted mark are reported separately
  1166→ * and are never counted as absent. A real integration queries by person
  1167→ * (not section) so a mid-term stream transfer still counts.
  1168→ */
  1169→export function attendanceRecordFor(studentId: string): AttendanceSummary {
  1170→  const cached = attendanceCache.get(studentId);
  1171→  if (cached) return cached;
  1172→  const cycle = currentCycle;
  1173→  const possible = possibleSchoolDays(cycle.attendanceFrom, cycle.attendanceTo);
  1174→  const rnd = mulberry32(hashStr(`attend|${studentId}`));
  1175→  const pinned = ATTENDANCE_PINNED[studentId] ?? {
  1176→    late: Math.floor(rnd() * 4), // 0-3 late arrivals
  1177→    absent: Math.floor(rnd() * 4), // 0-3 unexcused absences
  1178→    excused: Math.floor(rnd() * 3), // 0-2 excused
  1179→    notRecorded: Math.floor(rnd() * 3), // 0-2 days with no counted register
  1180→  };
  1181→  const recorded = possible - pinned.notRecorded;
  1182→  const attended = recorded - pinned.absent - pinned.excused;
  1183→  const present = Math.max(0, attended - pinned.late);
  1184→  // One entry per possible school day, in calendar order — the aggregator
  1185→  // only counts, so a stable placement keeps the demo deterministic.
  1186→  const marks: (AttendanceStatus | null)[] = [
  1187→    ...Array<AttendanceStatus>(present).fill("present"),
  1188→    ...Array<AttendanceStatus>(pinned.late).fill("late"),
  1189→    ...Array<AttendanceStatus>(pinned.absent).fill("absent"),
  1190→    ...Array<AttendanceStatus>(pinned.excused).fill("excused"),
  1191→    ...Array<AttendanceStatus | null>(pinned.notRecorded).fill(null),
  1192→  ];
  1193→  const summary = computeAttendanceSummary({
  1194→    from: cycle.attendanceFrom,
  1195→    to: cycle.attendanceTo,
  1196→    possibleSchoolDays: possible,
  1197→    marks,
  1198→  });
  1199→  attendanceCache.set(studentId, summary);
  1200→  return summary;
  1201→}
