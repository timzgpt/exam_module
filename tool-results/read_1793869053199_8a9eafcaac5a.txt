    63→interface PerformanceChartSectionProps {
    64→  /** Which report engine printed this card (mid-term vs terminal rules). */
    65→  variant: ChartVariant;
    66→  /** Static data-URI chart image; null renders the fallback paragraph. */
    67→  chartImage: string | null;

    69→  strongest: { name: string; score: number } | null;
    70→  /** Mid-Term primary cards only (chart prompt §8). */
    71→  bestImprovement: { name: string; value: number } | null;
    72→  isPrimary: boolean;
    73→}
    74→
    75→/**
    76→ * "Performance Charts" section (A4 section 6 / chart prompt §8): badge row,
    77→ * the static grouped bar chart image, the Terminal HTML legend and a
    78→ * visually hidden data table backing the chart for screen readers (B0:
    79→ * charts include a table fallback).
    80→ */
    81→function PerformanceChartSection({
    82→  variant,
    83→  chartImage,
    84→  rows,
    85→  strongest,
    86→  bestImprovement,
    87→  isPrimary,
    88→}: PerformanceChartSectionProps) {
    89→  // Section heading: "Performance Chart" on secondary terminal cards only.
    90→  const heading = variant === "mid-term" || isPrimary ? "Performance Charts" : "Performance Chart";
    91→  return (
    92→    <div className="mt-3 border border-border p-3">
    93→      <p className={CARD_LABEL}>{heading}</p>
    94→      {variant === "mid-term" ? (
    95→        <div className="mt-0.5">
    96→          <p className="text-compact font-medium">Subject Score Analysis</p>
    97→          <p className="text-2xs text-muted-foreground">
    98→            Student score vs class average across all subjects
    99→          </p>
   100→        </div>
   101→      ) : null}
   102→      {strongest || bestImprovement ? (
   103→        <div className="mt-2 flex flex-wrap items-center gap-1.5">
   104→          {strongest ? (
   105→            <span className="inline-flex items-center rounded-full border border-primary/25 bg-primary/8 px-3 py-1 text-2xs font-medium text-primary">
   106→              ★ Strongest Subject: {strongest.name} — {fmtTotalMark(strongest.score)} / 100
   107→            </span>
   108→          ) : null}
   109→          {bestImprovement ? (
   110→            <span className="inline-flex items-center rounded-full border border-success/30 bg-success/10 px-3 py-1 text-2xs font-medium text-success-strong">
   111→              ▲ Best Improvement: {bestImprovement.name} — +{bestImprovement.value}
   112→            </span>
   113→          ) : null}
   114→        </div>
   115→      ) : null}
   116→      {chartImage ? (
   117→        <img
   118→          src={chartImage}
   119→          alt="Grouped bar chart comparing achieved marks with the class average and class topper for every assessed subject"
   120→          className="chart-img mt-2 w-full"
   121→        />
   122→      ) : (
   123→        <p className="mt-2 text-2xs text-muted-foreground">{CHART_FALLBACK_TEXT}</p>
   124→      )}
   125→      {variant === "terminal" ? (
   126→        // Terminal legend is rendered in HTML, not in the chart image (§9).
   127→        <div className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 text-2xs text-muted-foreground">
   128→          <span className="inline-flex items-center gap-1">
   129→            <span className="size-2 rounded-sm" style={{ background: "#E74C3C" }} /> Achieved Marks
   130→          </span>
   131→          <span className="inline-flex items-center gap-1">
   132→            <span className="size-2 rounded-sm" style={{ background: "#3498DB" }} /> Class Avg
   133→          </span>
   134→          <span className="inline-flex items-center gap-1">
   135→            <span className="size-2 rounded-sm" style={{ background: "#2ECC71" }} /> Class Topper
   136→          </span>
   137→        </div>
   138→      ) : null}
   139→      {/* Chart data-table fallback (B0: charts include a table fallback). */}
   140→      <table className="sr-only">
   141→        <caption>Subject totals against the class average and class top</caption>
   142→        <thead>
   143→          <tr>
   144→            <th scope="col">Subject</th>
   145→            <th scope="col">Achieved</th>
   146→            <th scope="col">Class average</th>
   147→            <th scope="col">Class top</th>
   148→          </tr>
   149→        </thead>
   150→        <tbody>
   151→          {rows.map((row) => (
   152→            <tr key={row.name}>
   153→              <th scope="row">{row.name}</th>
   154→              <td>{fmtTotalMark(row.achieved)}</td>
   155→              <td>{fmtTotalMark(row.classAvg)}</td>
   156→              <td>{fmtTotalMark(row.classTop)}</td>
   157→            </tr>
   158→          ))}
   159→        </tbody>
   160→      </table>
   161→    </div>
   162→  );
   163→}

   165→export default function ReportCardPage() {
   166→  const params = useParams<{ studentId?: string }>();
   167→  const router = useRouter();
   168→  const { cycle } = useCycleContext();
   169→  const student = studentById(params.studentId ?? "") ?? studentById(DEMO_IDS.studentId)!;
   170→
   171→  const result = studentResultOf(student.id);
   172→  const classmates = studentsInClass(student.classId);
   173→  const classSize = classmates.length;
   174→  const attendance = attendanceRecordFor(student.id);
   175→  const comment = commentFor(student.classId, student.id);
   176→  const classTeacherName =
   177→    teacherById(classById(student.classId)?.classTeacherId)?.name ?? "\u2014";
   178→
   179→  const classSubjects = classSubjectsFor(student.classId);
   180→  const scheme = schemeForClass(student.classId);
   181→  const isPrimary = isPrimaryClass(student.classId);
   182→  // Assessment split shown in the table header (25 % monthly / 75 % terminal).
   183→  const caPct = classSubjects[0]?.caMarks ?? 25;
   184→  const examPct = classSubjects[0]?.examMarks ?? 75;
   185→  // Top band of the class scheme — rendered as the filled grade badge.
   186→  const topGrade = scheme.boundaries[0]?.grade;
   187→  const allRows = classSubjects.map((cs) => {
   188→    const subject = subjectById(cs.subjectId)!;
   189→    const comp = getComponents(student.classId, cs.subjectId)[student.id];
   190→    const enrolled = comp?.enrolled ?? true;
   191→    const mark = enrolled ? result?.marks[cs.subjectId] ?? null : null;
   192→    const grade = mark !== null ? gradeFor(mark, scheme) : undefined;
   193→    return { subject, enrolled, mark, grade, ca: comp?.ca ?? null, exam: comp?.exam ?? null };
   194→  });
   195→  // A subject with no mark at all (both components ABS/NA/NR/PENDING) is
   196→  // omitted from the card entirely; non-enrolled subsidiaries stay with "—".
   197→  // Remaining rows renumber continuously after any omitted subject (the #
   198→  // column below).
   199→  const rows = allRows.filter((r) => !r.enrolled || r.mark !== null);
   200→  const assessed = allRows.filter((r) => r.mark !== null);
   201→  const pointsTotal = assessed.reduce((a, r) => a + (r.grade?.points ?? 0), 0);
   202→  // Secondary division: best 7 subject points; null when fewer than 7 graded
   203→  // (and always null for primary classes — divisions are a NECTA outcome).
   204→  const division = isPrimary ? null : divisionFor(assessed.map((r) => r.grade?.points ?? null));
   205→  const passed = subjectsPassed(assessed.map((r) => r.grade?.grade ?? null));
   206→  // Strongest subject: highest total; ties go to the first in subject order.
   207→  const strongest =
   208→    assessed.length > 0
   209→      ? assessed.reduce((best, r) => ((r.mark as number) > (best.mark as number) ? r : best))
   210→      : undefined;
   211→  // Points and division are hidden on primary cards (Primary A–E bands carry
   212→  // no points) and on any secondary class whose teacher set the Report
   213→  // Overview "No Division" flag to Yes.
   214→  const noDivision = !isPrimary && (classById(student.classId)?.noDivision ?? false);
   215→  const showPoints = !isPrimary && !noDivision;
   216→  // Most improved (primary only): largest CA delta C2 − C1 per subject; a
   217→  // single recorded term counts as 0. Ties go to the first subject in order.
   218→  const mostImproved = (() => {
   219→    if (!isPrimary) return null;
   220→    const deltas = classSubjects.map((cs) => {
   221→      const comp = getComponents(student.classId, cs.subjectId)[student.id];
   222→      const cur = comp?.enrolled ? comp.ca ?? null : null;
   223→      const prev = prevCaOf(student.classId, cs.subjectId)[student.id] ?? null;
   224→      return { subjectId: cs.subjectId, delta: cur !== null && prev !== null ? cur - prev : 0 };
   225→    });
   226→    const best = mostImprovedSubject(deltas);
   227→    return best && best.delta > 0 ? best : null;
   228→  })();
   229→  const mostImprovedSubjectRow = mostImproved
   230→    ? rows.find((r) => r.subject.id === mostImproved.subjectId)
   231→    : undefined;
   232→
   233→  // Chart data (A4 section 6 / chart prompt §3–§4): achieved total vs class
   234→  // average and class top per assessed subject, over the student's enrolled
   235→  // classmates. Class stats follow the cycle's engine: mid-term counts a
   236→  // student with a single assessment at 100%; terminal only counts students
   237→  // with BOTH components (chart prompt §4/§9).
   238→  const variant: ChartVariant = cycle.kind === "mid-term" ? "mid-term" : "terminal";
   239→  const chartRows: ChartRow[] = assessed.map((r) => {
   240→    const cs = classSubjects.find((c) => c.subjectId === r.subject.id);
   241→    const comps = getComponents(student.classId, r.subject.id);
   242→    const scores = Object.values(comps)
   243→      .filter((c) => c.enrolled)
   244→      .map((c) => classScoreFor(variant, c.ca, c.exam, cs?.caMarks ?? 25, cs?.examMarks ?? 75))
   245→      .filter((t): t is number => t !== null);
   246→    return {
   247→      name: r.subject.name,
   248→      achieved: r.mark ?? 0,
   249→      classAvg: scores.length ? scores.reduce((a, b) => a + b, 0) / scores.length : 0,
   250→      classTop: scores.length ? Math.max(...scores) : 0,
   251→    };
   252→  });
   253→  // Series + static chart image (chart prompt §7 call sequence).
   254→  const chartSeries = buildChartSeries(
   255→    chartRows.map((r) => ({ name: r.name, achieved: r.achieved })),
   256→    chartRows.map((r) => r.classAvg),
   257→    chartRows.map((r) => r.classTop),
   258→    strongest?.subject.name ?? "",
   259→    variant,
   260→  );
   261→  const chartImage = generateChartImage(chartSeries, variant);
   262→  // Best-Improvement badge: Mid-Term primary cards only (chart prompt §8).
   263→  const chartBestImprovement =
   264→    variant === "mid-term" && mostImproved && mostImprovedSubjectRow
   265→      ? { name: mostImprovedSubjectRow.subject.name, value: mostImproved.delta }
   266→      : null;
   267→
   268→  const switchStudent = (id: string) => {
   269→    router.push(`/academic/report-cycles/${cycle.id}/students/${id}/report-card`);
   270→  };
   271→
   272→  const downloadPdf = usePrintDownload(
   273→    "Use 'Save as PDF' in the print dialog to download the report card.",
   274→  );

   276→  return (
   277→    <div className="space-y-5">
   278→      <PageHeader
   279→        title="Report card"
   280→        description={`${student.name} · ${classNameOf(student.classId)} · ${cycle.name}`}
   281→      />
   282→
   283→      {/* Toolbar: screen only, never printed */}
   284→      <div className="flex flex-col gap-3 rounded-lg border border-border/80 bg-white p-3 sm:flex-row sm:items-center sm:justify-between print:hidden">
   285→        <div className="flex flex-wrap items-center gap-2">
   286→          <span className="text-xs font-medium text-muted-foreground">Student</span>
   287→          {/* ds-allow-native-select: shadcn Select is a Radix listbox trigger, not a native select */}
   288→          <Select value={student.id} onValueChange={switchStudent}>
   289→            <SelectTrigger className="h-8 w-full text-compact sm:w-65">
   290→              <SelectValue />
   291→            </SelectTrigger>
   292→            <SelectContent>
   293→              {classmates.map((s) => (
   294→                <SelectItem key={s.id} value={s.id}>
   295→                  {s.name}
   296→                </SelectItem>
   297→              ))}
   298→            </SelectContent>
   299→          </Select>
   300→        </div>
   301→        <div className="flex items-center gap-2">
   302→          <Button size="sm" className="gap-1.5" onClick={downloadPdf}>
   303→            <Download className="size-4" /> Download PDF
   304→          </Button>
   305→          <Button size="sm" variant="outline" className="gap-1.5" onClick={() => window.print()}>
   306→            <Printer className="size-4" /> Print
   307→          </Button>
   308→        </div>
   309→      </div>
   310→
   311→      {/* Legacy rule 7: a student with no marks at all renders an empty
   312→          subject table — warn before printing instead of generating. */}
   313→      {assessed.length === 0 ? (
   314→        <div className="flex items-center gap-2 rounded-lg border border-destructive/40 bg-destructive/8 px-3 py-2 text-compact text-destructive print:hidden">
   315→          <AlertTriangle className="size-4 shrink-0" />
   316→          No marks have been entered for this student yet — report cards are generated only after marks exist.
   317→        </div>
   318→      ) : null}
   319→
   320→      {/* The printable card: scrollable on a phone, exact A4 sheet in print */}
   321→      <div className="overflow-x-auto print:overflow-visible">
   322→        <div
   323→          className="print-area mx-auto w-full max-w-205 border border-border bg-white p-8 font-serif text-foreground shadow-sm"
   324→        >
   325→        {/* School header */}
   326→        <p className="text-center text-xs font-semibold uppercase tracking-[0.35em]">
   327→          {SCHOOL.name}
   328→        </p>
   329→        <p className="mt-1 text-center text-xs italic text-muted-foreground">{SCHOOL.motto}</p>
   330→        <p className="mt-1 text-center text-2xs text-muted-foreground">
   331→          {SCHOOL.address} · {SCHOOL.phone}
   332→        </p>
   333→        <div className="mx-auto mt-3 h-1 max-w-md border-y border-input" />
   334→
   335→        <h2 className="mt-5 text-center text-base font-bold uppercase tracking-[0.18em]">
   336→          Student Report Card
   337→        </h2>
   338→        <p className="mt-0.5 text-center text-xs text-muted-foreground">
   339→          {cycle.term} · {cycle.academicYear} Academic Year
   340→        </p>
   341→        {/* Report date + school opens badges (A4 header; never hard-coded). */}
   342→        <div className="mt-3 flex flex-wrap items-center justify-center gap-2">
   343→          <span className="rounded border border-border px-2.5 py-1 text-2xs">
   344→            <span className="font-semibold uppercase tracking-[0.12em] text-muted-foreground">Report date</span>{" "}

   346→          </span>

   353→        {/* Student info grid */}

   387→        {/* Subjects table — the school's maroon ledger design: monthly and

   391→        <div className="mt-6 overflow-hidden rounded-lg bg-[#F7F2E9] font-sans">

   469→                      </td>

   475→                      >
