    28→    28→function hashStr(s: string): number {
    29→    29→  let h = 2166136261;
    30→    30→  for (let i = 0; i < s.length; i++) {
    31→    31→    h ^= s.charCodeAt(i);
    32→    32→    h = Math.imul(h, 16777619);
    33→    33→  }
    34→    34→  return h >>> 0;
    35→    35→}
    36→    36→
    37→    37→function mulberry32(seed: number): () => number {
    38→    38→  let a = seed;
    39→    39→  return () => {
    40→    40→    a |= 0;
    41→    41→    a = (a + 0x6d2b79f5) | 0;
    42→    42→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    43→    43→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    44→    44→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    45→    45→  };
    46→    46→}

   348→   348→/**
   349→   349→ * Is this student enrolled in the class subject? Core subjects are compulsory
   350→   350→ * (every student takes them); subsidiaries are opt-in per student, seeded so
   351→   351→ * every page sees the same enrolment.
   352→   352→ */
   353→   353→export function isEnrolledIn(
   354→   354→  classId: string,
   355→   355→  subjectId: string,
   356→   356→  studentId: string,
   357→   357→): boolean {
   358→   358→  if (!SUBSIDIARY_SUBJECT_IDS.has(subjectId)) return true;
   359→   359→  if (ENROL_OVERRIDES.has(`${studentId}|${subjectId}`)) return true;
   360→   360→  return hashStr(`enrol|${classId}|${subjectId}|${studentId}`) % 10 < 6;
   361→   361→}

   364→   364→
   365→   365→export function getMarks(
   366→   366→  classId: string,
   367→   367→  subjectId: string,
   368→   368→): Record<string, number | null> {
   369→   369→  const key = `${classId}|${subjectId}`;
   370→   370→  const cached = marksCache.get(key);
   371→   371→  if (cached) return cached;
   372→   372→
   373→   373→  const rnd = mulberry32(hashStr(key));
   374→   374→  const classOffset = (hashStr(classId) % 13) - 6;
   375→   375→  const subjectOffset = (hashStr(subjectId) % 15) - 7;
   376→   376→  const result: Record<string, number | null> = {};
   377→   377→  for (const s of studentsInClass(classId)) {
   378→   378→    if (!isEnrolledIn(classId, subjectId, s.id)) {
   379→   379→      result[s.id] = null; // not taking a subsidiary subject — dash everywhere
   380→   380→      continue;
   381→   381→    }
   382→   382→    const ability = 44 + (hashStr(s.id) % 46); // 44-89 latent ability
   383→   383→    const jitter = rnd() * 18 - 9;
   384→   384→    const ca = Math.max(0, Math.min(100, Math.round(ability + classOffset + jitter)));
   385→   385→    const exam = Math.max(0, Math.min(100, ca + Math.round(rnd() * 16 - 8)));
   386→   386→    // Subject total via the 30/70 weighting so components round-trip exactly.
   387→   387→    result[s.id] = subjectTotal(ca, exam, 30, 70);
   388→   388→  }
   389→   389→  const patch = NULL_PATCHES[key];
   390→   390→  if (patch === "ALL") {
   391→   391→    for (const sid of Object.keys(result)) result[sid] = null;
   392→   392→  } else if (patch) {
   393→   393→    for (const sid of patch) result[sid] = null;
   394→   394→  }
   395→   395→  for (const sid of Object.keys(result)) {
   396→   396→    if (DISQ_PATCHES.has(`${key}|${sid}`)) result[sid] = 0; // DISQ is a real zero
   397→   397→  }
   398→   398→  marksCache.set(key, result);
   399→   399→  return result;
   400→   400→}

   407→   407→const STATUS_PATCHES: Record<string, Record<string, MarkStatusCode>> = {
   408→   408→  "cls-4a|s5": { "f4a-03": "ABS", "f4a-07": "ABS", "f4a-10": "NA", "f4a-11": "NA" },
   409→   409→};
   410→   410→
   411→   411→/**
   412→   412→ * Status code typed in place of a mark. ABS/NA/NR/PENDING are treated as
   413→   413→ * blank (null); DISQ is a real zero. Non-enrolled subsidiary students show
   414→   414→ * "—" and never carry a status.
   415→   415→ */
   416→   416→export function markStatusOf(
   417→   417→  classId: string,
   418→   418→  subjectId: string,
   419→   419→  studentId: string,
   420→   420→): MarkStatusCode | null {
   421→   421→  if (`${classId}|${subjectId}` === "cls-2a|s7") return "PENDING";
   422→   422→  if (DISQ_PATCHES.has(`${classId}|${subjectId}|${studentId}`)) return "DISQ";
   423→   423→  return STATUS_PATCHES[`${classId}|${subjectId}`]?.[studentId] ?? null;
   424→   424→}
   425→   425→
   426→   426→export interface MarkComponents {
   427→   427→  enrolled: boolean;
   428→   428→  /** Continuous assessment, entered out of 100 (weighted 30%). */
   429→   429→  ca: number | null;
   430→   430→  /** End-of-term written exam, entered out of 100 (weighted 70%). */
   431→   431→  exam: number | null;
   432→   432→  status: MarkStatusCode | null;
   433→   433→}
   434→   434→
   435→   435→const componentsCache = new Map<string, Record<string, MarkComponents>>();
   436→   436→
   437→   437→/**
   438→   438→ * CA / Exam components per student. Components round-trip exactly to the
   439→   439→ * stored subject total via subjectTotal(ca, exam, 30, 70).
   440→   440→ */
   441→   441→export function getComponents(
   442→   442→  classId: string,
   443→   443→  subjectId: string,
   444→   444→): Record<string, MarkComponents> {
   445→   445→  const key = `${classId}|${subjectId}`;
   446→   446→  const cached = componentsCache.get(key);
   447→   447→  if (cached) return cached;
   448→   448→  const marks = getMarks(classId, subjectId);
   449→   449→  const result: Record<string, MarkComponents> = {};
   450→   450→  for (const [studentId, total] of Object.entries(marks)) {
   451→   451→    const enrolled = isEnrolledIn(classId, subjectId, studentId);
   452→   452→    const status = markStatusOf(classId, subjectId, studentId);
   453→   453→    if (total === null || !enrolled) {
   454→   454→      result[studentId] = { enrolled, ca: null, exam: null, status };
   455→   455→      continue;
   456→   456→    }
   457→   457→    // Invert the weighting exactly: search integer CA/Exam pairs near the
   458→   458→    // total until round(CA x 0.3 + Exam x 0.7, 2) reproduces the total.
   459→   459→    let ca = Math.max(0, Math.min(100, Math.round(total)));
   460→   460→    let exam = Math.max(0, Math.min(100, Math.round((total - 0.3 * ca) / 0.7)));
   461→   461→    for (let c = Math.max(0, Math.round(total) - 10); c <= Math.min(100, Math.round(total) + 10); c++) {
   462→   462→      const e = Math.round((total - 0.3 * c) / 0.7);
   463→   463→      if (e < 0 || e > 100) continue;
   464→   464→      if (Math.round((0.3 * c + 0.7 * e) * 100) / 100 === total) {
   465→   465→        ca = c;
   466→   466→        exam = e;
   467→   467→        break;
   468→   468→      }
   469→   469→    }
   470→   470→    result[studentId] = { enrolled, ca, exam, status };
   471→   471→  }
   472→   472→  componentsCache.set(key, result);
   473→   473→  return result;
   474→   474→}

   487→   487→ * ---------------------------------------------------------------- */
   488→   488→
   489→   489→const prevCaCache = new Map<string, Record<string, number | null>>();
   490→   490→
   491→   491→/**
   492→   492→ * Last term's continuous-assessment mark (C1) per student, used by the
   493→   493→ * Primary report card's most-improved calculation (C2 − C1). Seeded around
   494→   494→ * this term's CA so the delta lands in a plausible −10…+18 band. Students
   495→   495→ * without a previous record (new arrivals) get null — their delta counts as
   496→   496→ * 0, never as an improvement.
   497→   497→ */
   498→   498→export function prevCaOf(
   499→   499→  classId: string,
   500→   500→  subjectId: string,
   501→   501→): Record<string, number | null> {
   502→   502→  const key = `prevca|${classId}|${subjectId}`;
   503→   503→  const cached = prevCaCache.get(key);
   504→   504→  if (cached) return cached;
   505→   505→  const comps = getComponents(classId, subjectId);
   506→   506→  const result: Record<string, number | null> = {};
   507→   507→  for (const [studentId, comp] of Object.entries(comps)) {
   508→   508→    if (!comp.enrolled || comp.ca === null) {
   509→   509→      result[studentId] = null;
   510→   510→      continue;
   511→   511→    }
   512→   512→    const delta = (hashStr(`prev|${key}|${studentId}`) % 29) - 10; // −10…+18
   513→   513→    result[studentId] = Math.max(0, Math.min(100, comp.ca - delta));
   514→   514→  }
   515→   515→  prevCaCache.set(key, result);
   516→   516→  return result;
   517→   517→}

   876→   876→const classResultsCache = new Map<string, StudentResult[]>();
   877→   877→
   878→   878→export function classResults(classId: string): StudentResult[] {
   879→   879→  const cached = classResultsCache.get(classId);
   880→   880→  if (cached) return cached;
   881→   881→  const scheme = schemeForClass(classId);
   882→   882→  // Divisions are a secondary (NECTA CSEE) outcome — never computed for primary.
   883→   883→  const allowDivision = !isPrimaryClass(classId);
   884→   884→  const rows = studentsInClass(classId).map((student) => {
   885→   885→    const marks: Record<string, number | null> = {};
   886→   886→    const points: (number | null)[] = [];
   887→   887→    const grades: (string | null)[] = [];
   888→   888→    let total = 0;
   889→   889→    let count = 0;
   890→   890→    for (const sub of subjects) {
   891→   891→      const m = getMarks(classId, sub.id)[student.id];
   892→   892→      marks[sub.id] = m;
   893→   893→      if (m !== null) {
   894→   894→        total += m;
   895→   895→        count += 1;
   896→   896→        const g = gradeFor(m, scheme);
   897→   897→        grades.push(g?.grade ?? null);
   898→   898→        points.push(g?.points ?? null);
   899→   899→      } else {
   900→   900→        grades.push(null);
   901→   901→        points.push(null);
   902→   902→      }
   903→   903→    }
   904→   904→    const average = count > 0 ? total / count : 0;
   905→   905→    const division = allowDivision ? divisionFor(points) : null;
   906→   906→    return {
   907→   907→      student,
   908→   908→      marks,
   909→   909→      total,
   910→   910→      count,
   911→   911→      average,
   912→   912→      grade: count > 0 ? gradeFor(average, scheme)?.grade ?? scheme.boundaries[scheme.boundaries.length - 1].grade : "-",
   913→   913→      rank: 0,
   914→   914→      bestSevenPoints: division?.pointsSum ?? null,
   915→   915→      division,
   916→   916→      passedCount: subjectsPassed(grades),
   917→   917→    };
   918→   918→  });
   919→   919→  rows.sort((a, b) => b.average - a.average);
   920→   920→  rows.forEach((r, i) => (r.rank = i + 1));
   921→   921→  classResultsCache.set(classId, rows);
   922→   922→  return rows;
   923→   923→}

   936→   936→export function subjectAveragesFor(classId: string): {
   937→   937→  subjectId: string;
   938→   938→  avg: number;
   939→   939→  passRate: number;
   940→   940→  highest: number;
   941→   941→  lowest: number;

   947→   947→      (v): v is number => v !== null,
   948→   948→    );
   949→   949→    const avg = marks.length ? marks.reduce((a, b) => a + b, 0) / marks.length : 0;
   950→   950→    const distribution: Record<string, number> = {};
   951→   951→    for (const b of scheme.boundaries) distribution[b.grade] = 0;
   952→   952→    for (const m of marks) {
   953→   953→      const g = gradeFor(m, scheme)?.grade;
   954→   954→      if (g) distribution[g] += 1;
   955→   955→    }
   956→   956→    return {
   957→   957→      subjectId: cs.subjectId,
   958→   958→      avg,
   959→   959→      passRate: marks.length ? Math.round((marks.filter((m) => m >= scheme.passMark).length / marks.length) * 100) : 0,
   960→   960→      highest: marks.length ? Math.max(...marks) : 0,
   961→   961→      lowest: marks.length ? Math.min(...marks) : 0,
   962→   962→      distribution,
   963→   963→    };
   964→   964→  });

   967→   967→export interface CycleStats {
   968→   968→  totalStudents: number;
   969→   969→  average: number;
   970→   970→  passRate: number;
   971→   971→  gradeDistribution: Record<string, number>;
   972→   972→  classAverages: { classId: string; name: string; avg: number }[];
   973→   973→  subjectAverages: { subjectId: string; name: string; department: string; avg: number; passRate: number }[];
   974→   974→  marksCompletion: number;
   975→   975→  commentsCompletion: number;
   976→   976→  topStudents: { name: string; className: string; avg: number }[];
   977→   977→}
   978→   978→
   979→   979→const cycleStatsCache = new Map<string, CycleStats>();
   980→   980→
   981→   981→export function cycleStatsFor(cycleId: string): CycleStats {
   982→   982→  const cached = cycleStatsCache.get(cycleId);
   983→   983→  if (cached) return cached;
   984→   984→  const all = classes.flatMap((c) => classResults(c.id));
   985→   985→  const allMarks = all.flatMap((r) => Object.values(r.marks).filter((v): v is number => v !== null));
   986→   986→  const average = allMarks.length ? allMarks.reduce((a, b) => a + b, 0) / allMarks.length : 0;
   987→   987→  // Grade each class's marks with THAT class's scheme (primary A-E vs NECTA
   988→   988→  // CSEE A-F) so the school-wide distribution never mis-files a primary mark.
   989→   989→  const gradeDistribution: Record<string, number> = {};
   990→   990→  for (const s of gradingSchemes) for (const b of s.boundaries) gradeDistribution[b.grade] = 0;
   991→   991→  let passedEntries = 0;
   992→   992→  for (const c of classes) {
   993→   993→    const scheme = schemeForClass(c.id);
   994→   994→    for (const r of classResults(c.id)) {
   995→   995→      for (const m of Object.values(r.marks)) {
   996→   996→        if (m === null) continue;
   997→   997→        const g = gradeFor(m, scheme)?.grade;
   998→   998→        if (g) gradeDistribution[g] = (gradeDistribution[g] ?? 0) + 1;
   999→   999→        if (m >= scheme.passMark) passedEntries += 1;
  1000→  1000→      }
  1001→  1001→    }
  1002→  1002→  }
  1003→  1003→  const totalEntries = marksDeadlines.reduce((a, d) => a + d.entriesTotal, 0);
  1004→  1004→  const doneEntries = marksDeadlines.reduce((a, d) => a + d.entriesDone, 0);
  1005→  1005→  const commentTotals = classes.reduce(
  1006→  1006→    (acc, c) => {
  1007→  1007→      const p = commentProgress(c.id);
  1008→  1008→      return { done: acc.done + p.done, total: acc.total + p.total };
  1009→  1009→    },
  1010→  1010→    { done: 0, total: 0 },
  1011→  1011→  );
  1012→  1012→  const stats: CycleStats = {
  1013→  1013→    totalStudents: all.length,
  1014→  1014→    average,
  1015→  1015→    passRate: allMarks.length ? Math.round((passedEntries / allMarks.length) * 100) : 0,
  1016→  1016→    gradeDistribution,
  1017→  1017→    classAverages: classes.map((c) => ({
  1018→  1018→      classId: c.id,
  1019→  1019→      name: c.name,
  1020→  1020→      avg: classResults(c.id).reduce((a, r) => a + r.average, 0) / classResults(c.id).length,
  1021→  1021→    })),
  1022→  1022→    subjectAverages: subjects.map((sub) => {
  1023→  1023→      let entered = 0;
  1024→  1024→      let passed = 0;
  1025→  1025→      let sum = 0;
  1026→  1026→      for (const c of classes) {
  1027→  1027→        const scheme = schemeForClass(c.id);
  1028→  1028→        for (const m of Object.values(getMarks(c.id, sub.id))) {
  1029→  1029→          if (m === null) continue;
  1030→  1030→          entered += 1;
  1031→  1031→          sum += m;
  1032→  1032→          if (m >= scheme.passMark) passed += 1;
  1033→  1033→        }
  1034→  1034→      }
  1035→  1035→      return {
  1036→  1036→        subjectId: sub.id,
  1037→  1037→        name: sub.name,
  1038→  1038→        department: sub.department,
  1039→  1039→        avg: entered ? sum / entered : 0,
  1040→  1040→        passRate: entered ? Math.round((passed / entered) * 100) : 0,
  1041→  1041→      };
  1042→  1042→    }),
  1043→  1043→    marksCompletion: totalEntries ? Math.round((doneEntries / totalEntries) * 100) : 0,
  1044→  1044→    commentsCompletion: commentTotals.total ? Math.round((commentTotals.done / commentTotals.total) * 100) : 0,
  1045→  1045→    topStudents: all
  1046→  1046→      .slice()
  1047→  1047→      .sort((a, b) => b.average - a.average)
  1048→  1048→      .slice(0, 5)
  1049→  1049→      .map((r) => ({ name: r.student.name, className: classNameOf(r.student.classId), avg: r.average })),
  1050→  1050→  };
  1051→  1051→  cycleStatsCache.set(cycleId, stats);
  1052→  1052→  return stats;
  1053→  1053→}

  1067→  1067→  return deptNames.map((dept) => {
  1068→  1068→    const deptSubjects = subjects.filter((s) => s.department === dept);
  1069→  1069→    const rows = deptSubjects.map((sub) => {
  1070→  1070→      // Pass rate is judged against each class's own scheme pass mark.
  1071→  1071→      let entered = 0;
  1072→  1072→      let passed = 0;
  1073→  1073→      let sum = 0;
  1074→  1074→      for (const c of classes) {
  1075→  1075→        const scheme = schemeForClass(c.id);
  1076→  1076→        for (const m of Object.values(getMarks(c.id, sub.id))) {
  1077→  1077→          if (m === null) continue;
  1078→  1078→          entered += 1;
  1079→  1079→          sum += m;
  1080→  1080→          if (m >= scheme.passMark) passed += 1;
  1081→  1081→        }
  1082→  1082→      }
  1083→  1083→      return {
  1084→  1084→        subjectId: sub.id,
  1085→  1085→        name: sub.name,
  1086→  1086→        avg: entered ? sum / entered : 0,
  1087→  1087→        passRate: entered ? Math.round((passed / entered) * 100) : 0,
  1088→  1088→      };
  1089→  1089→    });
  1090→  1090→    const teacherRows = deptSubjects.flatMap((sub) => {
  1091→  1091→      const t = teachers.find((t) => t.subjectIds.includes(sub.id));
  1092→  1092→      if (!t) return [];
  1093→  1093→      const rel = marksDeadlines.filter((d) => d.subjectId === sub.id);
  1094→  1094→      const done = rel.reduce((a, d) => a + d.entriesDone, 0);
  1095→  1095→      const total = rel.reduce((a, d) => a + d.entriesTotal, 0);
  1096→  1096→      return [{ name: t.name, subject: sub.name, done, total }];
  1097→  1097→    });
  1098→  1098→    const submissionTotal = teacherRows.reduce((a, r) => a + r.total, 0);
  1099→  1099→    const submissionDone = teacherRows.reduce((a, r) => a + r.done, 0);
  1100→  1100→    const hod = teachers.find((t) => t.role.includes(`HOD ${dept}`));
  1101→  1101→    return {
  1102→  1102→      department: dept,
  1103→  1103→      hodName: hod?.name ?? "-",
  1104→  1104→      avg: rows.reduce((a, r) => a + r.avg, 0) / rows.length,
  1105→  1105→      passRate: Math.round(rows.reduce((a, r) => a + r.passRate, 0) / rows.length),
  1106→  1106→      submissionRate: submissionTotal ? Math.round((submissionDone / submissionTotal) * 100) : 0,
  1107→  1107→      subjects: rows,
  1108→  1108→      teachers: teacherRows,
  1109→  1109→    };
  1110→  1110→  });
  1111→  1111→}
