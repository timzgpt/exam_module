    26→    26→function hashStr(s: string): number {
    27→    27→  let h = 2166136261;
    28→    28→  for (let i = 0; i < s.length; i++) {
    29→    29→    h ^= s.charCodeAt(i);
    30→    30→    h = Math.imul(h, 16777619);
    31→    31→  }
    32→    32→  return h >>> 0;
    33→    33→}
    34→    34→
    35→    35→function mulberry32(seed: number): () => number {
    36→    36→  let a = seed;
    37→    37→  return () => {
    38→    38→    a |= 0;
    39→    39→    a = (a + 0x6d2b79f5) | 0;
    40→    40→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    41→    41→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    42→    42→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    43→    43→  };
    44→    44→}

   306→   306→const NULL_PATCHES: Record<string, string[] | "ALL"> = {
   307→   307→  "cls-4a|s5": ["f4a-03", "f4a-07", "f4a-10", "f4a-11"],

   313→   313→
   314→   314→export function getMarks(
   315→   315→  classId: string,
   316→   316→  subjectId: string,
   317→   317→): Record<string, number | null> {
   318→   318→  const key = `${classId}|${subjectId}`;
   319→   319→  const cached = marksCache.get(key);
   320→   320→  if (cached) return cached;
   321→   321→
   322→   322→  const rnd = mulberry32(hashStr(key));
   323→   323→  const classOffset = (hashStr(classId) % 13) - 6;
   324→   324→  const subjectOffset = (hashStr(subjectId) % 15) - 7;
   325→   325→  const result: Record<string, number | null> = {};
   326→   326→  for (const s of studentsInClass(classId)) {
   327→   327→    const ability = 44 + (hashStr(s.id) % 46); // 44 – 89 latent ability
   328→   328→    const jitter = rnd() * 18 - 9;
   329→   329→    const mark = Math.round(ability + classOffset + subjectOffset + jitter);
   330→   330→    result[s.id] = Math.max(14, Math.min(99, mark));
   331→   331→  }
   332→   332→  const patch = NULL_PATCHES[key];
   333→   333→  if (patch === "ALL") {
   334→   334→    for (const sid of Object.keys(result)) result[sid] = null;
   335→   335→  } else if (patch) {
   336→   336→    for (const sid of patch) result[sid] = null;
   337→   337→  }
   338→   338→  marksCache.set(key, result);
   339→   339→  return result;
   340→   340→}

   393→   393→export function commentFor(classId: string, studentId: string): CommentRecord {
   394→   394→  const rnd = mulberry32(hashStr(`comment|${classId}|${studentId}`));
   395→   395→  const missing = COMMENTS_MISSING.has(studentId);
   396→   396→  const draft = COMMENTS_DRAFT.has(studentId);

   400→   400→    studentId,
   401→   401→    classTeacher: missing ? null : commentTemplates[idx],
   402→   402→    headTeacher: missing || rnd() > 0.75 ? null : commentTemplates[second],

   705→   705→        total += m;
   706→   706→        count += 1;
   707→   707→      }
   708→   708→    }

   711→   711→  });
   712→   712→  rows.sort((a, b) => b.average - a.average);
   713→   713→  rows.forEach((r, i) => (r.rank = i + 1));
