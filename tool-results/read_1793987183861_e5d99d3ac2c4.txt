    32→/**
    33→ * Persisted store for the Academic Management module (task §10). All records
    34→ * are relational rows (see academic-admin.ts); actions enforce the module's
    35→ * governance rules (task §13): only Academic Masters and Administrators may
    36→ * mutate configuration, and every mutation appends an audit entry.
    37→ *
    38→ * The store seeds itself from buildSeedAcademicState() on first use — with
    39→ * the seed intact every derived view matches the existing demo behaviour.
    40→ */
    41→
    42→const MANAGE_ROLES = new Set(["academic-master", "administrator"]);
    43→
    44→export function currentActor(): string {
    45→  const v = useViewer.getState();
    46→  if (v.role === "administrator") return "Administrator";
    47→  if (v.role === "subject-teacher" && v.teacherId)
    48→    return `subject-teacher:${v.teacherId}`;
    49→  return "academic-master";
    50→}
    51→
    52→function audit(
    53→  data: AcademicAdminData,
    54→  action: string,
    55→  entity: string,
    56→  entityId: string,
    57→  detail: string,
    58→): AuditEntry {
    59→  return {
    60→    id: `audit-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
    61→    at: new Date().toISOString(),
    62→    actor: currentActor(),
    63→    action,
    64→    entity,
    65→    entityId,
    66→    detail,
    67→  };
    68→}
    69→
    70→/** Guard used by every mutating action. Returns an error toast message or null. */
    71→function guard(role?: string): string | null {
    72→  const r = role ?? useViewer.getState().role;
    73→  if (!MANAGE_ROLES.has(r))
    74→    return "Only an Academic Master or Administrator can change academic configuration.";
    75→  return null;
    76→}

   172→
   173→export const useAcademicAdminStore = create<AcademicAdminState>()(
   174→  persist(
   175→    (set, get) => ({
   176→      ...seed,
   177→
   178→      resetToSeed: () => set({ ...seed, audit: [...seed.audit] }),
   179→
   180→      /* ── catalogue ─────────────────────────────────────────────── */
   181→      registerItem: (draft) => {
   182→        const denied = guard();
   183→        if (denied) return { ok: false, error: denied };
   184→        const data = get();
   185→        const errors = validateItem(data, draft);
   186→        if (errors.length > 0) return { ok: false, error: errors.join(" ") };
   187→        const now = new Date().toISOString();
   188→        const id = `item-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
   189→        const item: AcademicItem = { ...draft, id, createdAt: now, updatedAt: now };
   190→        set((s) => ({
   191→          items: [...s.items, item],
   192→          audit: [...s.audit, audit(s, "register-item", "academic_item", id, `${item.name} (${item.code}) registered`)],
   193→        }));
   194→        return { ok: true };
   195→      },
   196→
   197→      updateItem: (id, patch) => {
   198→        const denied = guard();
   199→        if (denied) return { ok: false, error: denied };
   200→        const data = get();
   201→        const existing = itemById(data, id);
   202→        if (!existing) return { ok: false, error: "Item not found." };
   203→        const merged = { ...existing, ...patch };
   204→        const errors = validateItem(
   205→          data,
   206→          {
   207→            name: merged.name,
   208→            code: merged.code,
   209→            type: merged.type,
   210→            sections: merged.sections,
   211→            category: merged.category,
   212→          },
   213→          id,
   214→        );
   215→        if (errors.length > 0) return { ok: false, error: errors.join(" ") };
   216→        const now = new Date().toISOString();
   217→        set((s) => ({
   218→          items: s.items.map((i) =>
   219→            i.id === id ? { ...i, ...patch, updatedAt: now } : i,
   220→          ),
   221→          audit: [...s.audit, audit(s, "update-item", "academic_item", id, `${merged.name} updated`)],
   222→        }));
   223→        return { ok: true };
   224→      },
   225→
   226→      archiveItem: (id) => {
   227→        const denied = guard();
   228→        if (denied) return { ok: false, error: denied };
   229→        const item = itemById(get(), id);
   230→        if (!item) return { ok: false, error: "Item not found." };
   231→        const nextStatus = item.status === "active" ? "inactive" : "active";
   232→        // Business rule (task §4): archiving is always allowed; an inactive
   233→        // item cannot be ADDED to new curricula, historical records stay.
   234→        set((s) => ({
   235→          items: s.items.map((i) =>
   236→            i.id === id ? { ...i, status: nextStatus, updatedAt: new Date().toISOString() } : i,
   237→          ),
   238→          audit: [
   239→            ...s.audit,
   240→            audit(
   241→              s,
   242→              nextStatus === "inactive" ? "archive-item" : "restore-item",
   243→              "academic_item",
   244→              id,
   245→              `${item.name} ${nextStatus === "inactive" ? "archived" : "restored"}`,
   246→            ),
   247→          ],
   248→        }));
   249→        return { ok: true };
   250→      },
   251→
   252→      /* ── curricula ─────────────────────────────────────────────── */
   253→      createCurriculum: (draft) => {
   254→        const denied = guard();
   255→        if (denied) return { ok: false, error: denied };
   256→        const data = get();
   257→        const clash = data.curricula.find(
   258→          (c) =>
   259→            c.section === draft.section &&
   260→            c.level === draft.level &&
   261→            c.academicYear === draft.academicYear,
   262→        );
   263→        if (clash)
   264→          return {
   265→            ok: false,
   266→            error: `A ${draft.section} curriculum for ${draft.level} (${draft.academicYear}) already exists.`,
   267→          };
   268→        const id = `cur-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
   269→        const curriculum: Curriculum = {
   270→          ...draft,
   271→          id,
   272→          items: [],
   273→          version: 1,
   274→          status: "draft",
   275→          updatedAt: new Date().toISOString(),
   276→        };
   277→        set((s) => ({
   278→          curricula: [...s.curricula, curriculum],
   279→          audit: [...s.audit, audit(s, "create-curriculum", "curriculum", id, `${curriculum.name} created`)],
   280→        }));
   281→        return { ok: true, curriculumId: id };
   282→      },
   283→
   284→      addItemToCurriculum: (curriculumId, itemId, classification) => {
   285→        const denied = guard();
   286→        if (denied) return { ok: false, error: denied };
   287→        const data = get();
   288→        const curriculum = data.curricula.find((c) => c.id === curriculumId);
   289→        const item = itemById(data, itemId);
   290→        if (!curriculum || !item) return { ok: false, error: "Curriculum or item not found." };
   291→        if (curriculum.items.some((ci) => ci.itemId === itemId))
   292→          return { ok: false, error: `${item.name} is already in this curriculum.` };
   293→        if (item.status === "inactive")
   294→          return { ok: false, error: `${item.name} is archived and cannot be added to a new curriculum.` };
   295→        if (!item.sections.includes(curriculum.section))
   296→          return {
   297→            ok: false,
   298→            error: `${item.name} is not offered for the ${curriculum.section} section.`,
   299→          };
   300→        set((s) => ({
   301→          curricula: s.curricula.map((c) =>
   302→            c.id === curriculumId
   303→              ? {
   304→                  ...c,
   305→                  version: c.version + 1,
   306→                  status: "draft",
   307→                  updatedAt: new Date().toISOString(),
   308→                  items: [
   309→                    ...c.items,
   310→                    {
   311→                      itemId,
   312→                      classification,
   313→                      order: c.items.length + 1,
   314→                      periodsPerWeek: 3,
   315→                      examPapers: 1 as const,
   316→                      chartLabel: item.code.slice(0, 8),
   317→                      assessment: {
   318→                        caMarks: 25,
   319→                        examMarks: 75,
   320→                        fullMarks: 100,
   321→                        developmentalModel:
   322→                          c.section === "Early Years"
   323→                            ? ["Emerging", "Developing", "Achieved", "Exceeding Expectations"]
   324→                            : null,
   325→                      },
   326→                    },
   327→                  ],
   328→                }
   329→              : c,
   330→          ),
   331→          audit: [...s.audit, audit(s, "add-curriculum-item", "curriculum", curriculumId, `${item.name} added as ${classification}`)],
   332→        }));
   333→        return { ok: true };
   334→      },
   335→
   336→      removeItemFromCurriculum: (curriculumId, itemId) => {
   337→        const denied = guard();
   338→        if (denied) return { ok: false, error: denied };
   339→        const data = get();
   340→        const curriculum = data.curricula.find((c) => c.id === curriculumId);
   341→        const item = itemById(data, itemId);
   342→        if (!curriculum || !item) return { ok: false, error: "Curriculum or item not found." };
   343→        // Guard: refuse when the item has registration / assignment history in
   344→        // classes that adopted this curriculum (task §4 deletion restriction).
   345→        const adoptingClasses = Object.values(data.adoptions)
   346→          .filter((a) => a.curriculumId === curriculumId)
   347→          .map((a) => a.classId);
   348→        for (const classId of adoptingClasses) {
   349→          if (itemHasHistory(data, itemId, classId))
   350→            return {
   351→              ok: false,
   352→              error: `${item.name} has registration or teacher history in an adopting class — remove it from the class allocation instead.`,
   353→            };
   354→        }
   355→        set((s) => ({
   356→          curricula: s.curricula.map((c) =>
   357→            c.id === curriculumId
   358→              ? {
   359→                  ...c,
   360→                  version: c.version + 1,
   361→                  status: "draft",
   362→                  updatedAt: new Date().toISOString(),
   363→                  items: c.items
   364→                    .filter((ci) => ci.itemId !== itemId)
   365→                    .map((ci, i) => ({ ...ci, order: i + 1 })),
   366→                }
   367→              : c,
   368→          ),
   369→          audit: [...s.audit, audit(s, "remove-curriculum-item", "curriculum", curriculumId, `${item.name} removed`)],
   370→        }));
   371→        return { ok: true };
   372→      },
   373→
   374→      reorderCurriculumItem: (curriculumId, itemId, dir) => {
   375→        const denied = guard();
   376→        if (denied) return { ok: false, error: denied };
   377→        set((s) => ({
   378→          curricula: s.curricula.map((c) => {
   379→            if (c.id !== curriculumId) return c;
   380→            const items = [...c.items].sort((a, b) => a.order - b.order);
   381→            const idx = items.findIndex((ci) => ci.itemId === itemId);
   382→            const target = idx + dir;
   383→            if (idx < 0 || target < 0 || target >= items.length) return c;
   384→            [items[idx], items[target]] = [items[target], items[idx]];
   385→            return {
   386→              ...c,
   387→              version: c.version + 1,
   388→              updatedAt: new Date().toISOString(),
   389→              items: items.map((ci, i) => ({ ...ci, order: i + 1 })),
   390→            };
   391→          }),
   392→        }));
   393→        return { ok: true };
   394→      },
   395→
   396→      updateCurriculumItem: (curriculumId, itemId, patch) => {
   397→        const denied = guard();
   398→        if (denied) return { ok: false, error: denied };
   399→        set((s) => ({
   400→          curricula: s.curricula.map((c) =>
   401→            c.id === curriculumId
   402→              ? {
   403→                  ...c,
   404→                  version: c.version + 1,
   405→                  updatedAt: new Date().toISOString(),
   406→                  items: c.items.map((ci) =>
   407→                    ci.itemId === itemId ? { ...ci, ...patch } : ci,
   408→                  ),
   409→                }
   410→              : c,
   411→          ),
   412→        }));
   413→        return { ok: true };
   414→      },
   415→
   416→      publishCurriculum: (curriculumId) => {
   417→        const denied = guard();
   418→        if (denied) return { ok: false, error: denied };
   419→        const curriculum = get().curricula.find((c) => c.id === curriculumId);
   420→        if (!curriculum) return { ok: false, error: "Curriculum not found." };
   421→        if (curriculum.items.length === 0)
   422→          return { ok: false, error: "Add at least one item before publishing." };
   423→        set((s) => ({
   424→          curricula: s.curricula.map((c) =>
   425→            c.id === curriculumId
   426→              ? { ...c, status: "published", updatedAt: new Date().toISOString() }
   427→              : c,
   428→          ),
   429→          audit: [...s.audit, audit(s, "publish-curriculum", "curriculum", curriculumId, `${curriculum.name} published (v${curriculum.version + 1})`)],
   430→        }));
   431→        return { ok: true };
   432→      },
   433→
   434→      /* ── combinations ──────────────────────────────────────────── */
   435→      saveCombination: (draft) => {
   436→        const denied = guard();
   437→        if (denied) return { ok: false, error: denied };
   438→        if (!draft.name.trim() || !draft.code.trim())
   439→          return { ok: false, error: "Combination name and code are required." };
   440→        if (draft.itemIds.length === 0)
   441→          return { ok: false, error: "Add at least one optional subject." };
   442→        set((s) => {
   443→          if (draft.id && s.combinations.some((c) => c.id === draft.id)) {
   444→            return {
   445→              combinations: s.combinations.map((c) =>
   446→                c.id === draft.id ? { ...c, ...draft, id: c.id } : c,
   447→              ),
   448→              audit: [...s.audit, audit(s, "update-combination", "combination", draft.id, `${draft.name} updated`)],
   449→            };
   450→          }
   451→          const id = `combo-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
   452→          return {
   453→            combinations: [...s.combinations, { ...draft, id }],
   454→            audit: [...s.audit, audit(s, "create-combination", "combination", id, `${draft.name} created`)],
   455→          };
   456→        });
   457→        return { ok: true };
   458→      },
   459→
   460→      removeCombination: (id) => {
   461→        const denied = guard();
   462→        if (denied) return { ok: false, error: denied };
   463→        const inUse = get().registrations.some(
   464→          (r) => r.combinationId === id && r.status === "active",
   465→        );
   466→        if (inUse)
   467→          return { ok: false, error: "Students are registered through this combination — reassign them first." };
   468→        set((s) => ({
   469→          combinations: s.combinations.filter((c) => c.id !== id),
   470→          audit: [...s.audit, audit(s, "remove-combination", "combination", id, "removed")],
   471→        }));
   472→        return { ok: true };
   473→      },
   474→
   475→      /* ── class allocation ──────────────────────────────────────── */
   476→      adoptCurriculum: (classId, academicYear, curriculumId) => {
   477→        const denied = guard();
   478→        if (denied) return { ok: false, error: denied };
   479→        const curriculum = get().curricula.find((c) => c.id === curriculumId);
   480→        if (!curriculum) return { ok: false, error: "Curriculum not found." };
   481→        const adoption: ClassCurriculumAdoption = {
   482→          classId,
   483→          academicYear,
   484→          curriculumId,
   485→          status: "draft",
   486→          publishedAt: null,
   487→        };
   488→        set((s) => ({
   489→          adoptions: { ...s.adoptions, [adoptionKey(classId, academicYear)]: adoption },
   490→          audit: [...s.audit, audit(s, "adopt-curriculum", "class_curriculum", adoptionKey(classId, academicYear), `${classId} adopted ${curriculum.name}`)],
   491→        }));
   492→        return { ok: true };
   493→      },
   494→
   495→      bulkAdoptCurriculum: (classIds, academicYear, curriculumId) => {
   496→        const denied = guard();
   497→        if (denied) return { ok: false, error: denied };
   498→        const curriculum = get().curricula.find((c) => c.id === curriculumId);
   499→        if (!curriculum) return { ok: false, error: "Curriculum not found." };
   500→        set((s) => {
   501→          const adoptions = { ...s.adoptions };
   502→          for (const classId of classIds)
   503→            adoptions[adoptionKey(classId, academicYear)] = {
   504→              classId,
   505→              academicYear,
   506→              curriculumId,
   507→              status: "draft",
   508→              publishedAt: null,
   509→            };
   510→          return {
   511→            adoptions,
   512→            audit: [
   513→              ...s.audit,
   514→              audit(
   515→                s,
   516→                "bulk-adopt-curriculum",
   517→                "class_curriculum",
   518→                curriculumId,
   519→                `${classIds.length} classes adopted ${curriculum.name}`,
   520→              ),
   521→            ],
   522→          };
   523→        });
   524→        return { ok: true };
   525→      },
   526→
   527→      publishAdoption: (classId, academicYear) => {
   528→        const denied = guard();
   529→        if (denied) return { ok: false, error: denied };
   530→        const key = adoptionKey(classId, academicYear);
   531→        const adoption = get().adoptions[key];
   532→        if (!adoption) return { ok: false, error: "The class has no adopted curriculum." };
   533→        set((s) => ({
   534→          adoptions: {
   535→            ...s.adoptions,
   536→            [key]: { ...adoption, status: "published", publishedAt: new Date().toISOString() },
   537→          },
   538→          audit: [...s.audit, audit(s, "publish-allocation", "class_curriculum", key, `${classId} allocation published`)],
   539→        }));
   540→        return { ok: true };
   541→      },
   542→
   543→      removeFromAdoption: (classId, academicYear, itemId) => {
   544→        const denied = guard();
   545→        if (denied) return { ok: false, error: denied };
   546→        const data = get();
   547→        if (itemHasHistory(data, itemId, classId))
   548→          return {
   549→            ok: false,
   550→            error: "This item has registration or teacher history in the class — drop the registrations first.",
   551→          };
   552→        const key = adoptionKey(classId, academicYear);
   553→        const adoption = data.adoptions[key];
   554→        if (!adoption) return { ok: false, error: "The class has no adopted curriculum." };
   555→        set((s) => {
   556→          const curriculum = s.curricula.find((c) => c.id === adoption.curriculumId);
   557→          if (!curriculum) return {};
   558→          return {
   559→            curricula: s.curricula.map((c) =>
   560→              c.id === curriculum.id
   561→                ? {
   562→                    ...c,
   563→                    version: c.version + 1,
   564→                    updatedAt: new Date().toISOString(),
   565→                    items: c.items
   566→                      .filter((ci) => ci.itemId !== itemId)
   567→                      .map((ci, i) => ({ ...ci, order: i + 1 })),
   568→                  }
   569→                : c,
   570→            ),
   571→            audit: [
   572→              ...s.audit,
   573→              audit(
   574→                s,
   575→                "remove-allocation-item",
   576→                "class_curriculum",
   577→                key,
   578→                `${itemById(data, itemId)?.name ?? itemId} removed from ${classId}`,
   579→              ),
   580→            ],
   581→          };
   582→        });
   583→        return { ok: true };
   584→      },
   585→
   586→      setAssessmentOverride: (classId, itemId, config) => {
   587→        const denied = guard();
   588→        if (denied) return { ok: false, error: denied };
   589→        if (config.caMarks + config.examMarks <= 0)
   590→          return { ok: false, error: "CA + Exam marks must add up to more than 0." };
   591→        set((s) => ({
   592→          assessmentOverrides: {
   593→            ...s.assessmentOverrides,
   594→            [assessmentKey(classId, itemId)]: {
   595→              caMarks: config.caMarks,
   596→              examMarks: config.examMarks,
   597→              fullMarks: config.caMarks + config.examMarks,
   598→            },
   599→          },
   600→          audit: [
   601→            ...s.audit,
   602→            audit(

   604→              "configure-assessment",

   608→            ),
   609→          ],
   610→        }));

   612→      },

   614→      clearAssessmentOverride: (classId, itemId) => {
   615→        const denied = guard();
   616→        if (denied) return { ok: false, error: denied };
   617→        set((s) => {
   618→          const overrides = { ...s.assessmentOverrides };
   619→          delete overrides[assessmentKey(classId, itemId)];
   620→          return { assessmentOverrides: overrides };
   621→        });
   622→        return { ok: true };
   623→      },
   624→
   625→      setRegistrationMode: (classId, year, mode) => {
   626→        const denied = guard();
   627→        if (denied) return { ok: false, error: denied };
   628→        set((s) => ({
   629→          registrationModes: { ...s.registrationModes, [adoptionKey(classId, year)]: mode },
   630→          audit: [...s.audit, audit(s, "set-registration-mode", "registration_mode", adoptionKey(classId, year), mode)],
   631→        }));
   632→        return { ok: true };
   633→      },
   634→
   635→      /* ── teacher allocation ────────────────────────────────────── */
   636→      assignTeacher: (draft) => {
   637→        const denied = guard();
   638→        if (denied) return { ok: false, error: denied };
   639→        const data = get();
   640→        const duplicate = data.assignments.find(
   641→          (a) =>
   642→            a.classId === draft.classId &&
   643→            a.itemId === draft.itemId &&
   644→            a.teacherId === draft.teacherId &&
   645→            a.status === "assigned" &&
   646→            a.endsAt === null &&
   647→            a.academicYear === draft.academicYear,
   648→        );
   649→        if (duplicate)
   650→          return { ok: false, error: "Duplicate assignment: this teacher already holds this class subject." };
   651→        const now = new Date().toISOString();
   652→        // Reassignment history: end the previous active assignment, keep the row.
   653→        const previous = data.assignments.find(
   654→          (a) =>
   655→            a.classId === draft.classId &&
   656→            a.itemId === draft.itemId &&
   657→            a.status === "assigned" &&
   658→            a.endsAt === null &&
   659→            a.academicYear === draft.academicYear,
   660→        );
   661→        const row: TeacherItemAssignment = {
   662→          ...draft,
   663→          id: `ta-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   664→          status: draft.status ?? "assigned",
   665→          assignedAt: now,
   666→          endsAt: null,
   667→        };
   668→        set((s) => ({
   669→          assignments: [
   670→            ...s.assignments.map((a) =>
   671→              previous && a.id === previous.id ? { ...a, endsAt: now } : a,
   672→            ),
   673→            row,
   674→          ],
   675→          audit: [
   676→            ...s.audit,
   677→            audit(
   678→              s,
   679→              "assign-teacher",
   680→              "teacher_assignment",
   681→              row.id,
   682→              `${row.teacherId} → ${row.classId} · ${itemById(data, row.itemId)?.name ?? row.itemId}`,
   683→            ),
   684→          ],
   685→        }));
   686→        return { ok: true };
   687→      },
   688→
   689→      unassignTeacher: (assignmentId) => {
   690→        const denied = guard();
   691→        if (denied) return { ok: false, error: denied };
   692→        const row = get().assignments.find((a) => a.id === assignmentId);
   693→        if (!row) return { ok: false, error: "Assignment not found." };
   694→        set((s) => ({
   695→          assignments: s.assignments.map((a) =>
   696→            a.id === assignmentId ? { ...a, endsAt: new Date().toISOString() } : a,
   697→          ),
   698→          audit: [...s.audit, audit(s, "unassign-teacher", "teacher_assignment", assignmentId, `${row.teacherId} released from ${row.classId}`)],
   699→        }));
   700→        return { ok: true };
   701→      },
   702→
   703→      /* ── student registration ──────────────────────────────────── */
   704→      registerStudentSubjects: (studentId, classId, optionalItemIds, opts) => {
   705→        const denied = guard();
   706→        if (denied) return { ok: false, error: denied };
   707→        const data = get();
   708→        const year = "2026";
   709→        const validation = validateRegistration(
   710→          data,
   711→          studentId,
   712→          classId,
   713→          optionalItemIds,
   714→          opts?.combinationId ?? null,
   715→          year,
   716→        );
   717→        if (validation.errors.length > 0)
   718→          return { ok: false, error: validation.errors.join(" ") };
   719→        const now = new Date().toISOString();
   720→        const rows: StudentSubjectRegistration[] = optionalItemIds.map((itemId) => ({
   721→          id: `reg-${classId}-${studentId}-${itemId}`,
   722→          studentId,
   723→          classId,
   724→          itemId,
   725→          academicYear: year,
   726→          classification: "optional" as const,
   727→          source: opts?.source ?? "individual",
   728→          combinationId: opts?.combinationId ?? null,
   729→          status: "active" as const,
   730→          registeredAt: now,
   731→          droppedAt: null,
   732→        }));
   733→        set((s) => ({
   734→          registrations: [
   735→            ...s.registrations.filter(
   736→              (r) => !rows.some((row) => row.id === r.id),
   737→            ),
   738→            ...rows,
   739→          ],
   740→          audit: [
   741→            ...s.audit,
   742→            audit(
   743→              s,
   744→              "register-subjects",
   745→              "student_registration",
   746→              `${classId}|${studentId}`,
   747→              `${rows.length} optional subject(s) registered${opts?.combinationId ? ` via ${opts.combinationId}` : ""}`,
   748→            ),
   749→          ],
   750→        }));
   751→        return { ok: true };
   752→      },
   753→
   754→      dropRegistration: (registrationId) => {
   755→        const denied = guard();
   756→        if (denied) return { ok: false, error: denied };
   757→        const row = get().registrations.find((r) => r.id === registrationId);
   758→        if (!row) return { ok: false, error: "Registration not found." };
   759→        set((s) => ({
   760→          registrations: s.registrations.map((r) =>
   761→            r.id === registrationId
   762→              ? { ...r, status: "dropped", droppedAt: new Date().toISOString() }
   763→              : r,
   764→          ),
   765→          audit: [
   766→            ...s.audit,
   767→            audit(
   768→              s,
   769→              "drop-registration",
   770→              "student_registration",
   771→              registrationId,
   772→              `${row.itemId} dropped for ${row.studentId} (history kept)`,
   773→            ),
   774→          ],
   775→        }));
   776→        return { ok: true };
   777→      },
   778→
   779→      bulkRegisterClass: (classId, year, opts) => {
   780→        const denied = guard();
   781→        if (denied) return { ok: false, error: denied };
   782→        const data = get();
   783→        const { adoptedCurriculum, studentsInClassOf } = bulkHelpers(data, classId, year);
   784→        if (!adoptedCurriculum) return { ok: false, error: "The class has no adopted curriculum." };
   785→        const now = new Date().toISOString();
   786→        const rows: StudentSubjectRegistration[] = [];
   787→        for (const student of studentsInClassOf) {
   788→          for (const ci of adoptedCurriculum.items) {
   789→            if (ci.classification === "core") {
   790→              rows.push({
   791→                id: `reg-${classId}-${student.id}-${ci.itemId}`,
   792→                studentId: student.id,
   793→                classId,
   794→                itemId: ci.itemId,
   795→                academicYear: year,
   796→                classification: "core",
   797→                source: "auto-core",
   798→                combinationId: null,
   799→                status: "active",
   800→                registeredAt: now,
   801→                droppedAt: null,
   802→              });
   803→            }
   804→          }
   805→          const combinationId = opts?.combinationIdByStudent?.[student.id] ?? null;
   806→          if (combinationId) {
   807→            const combo = data.combinations.find((c) => c.id === combinationId);
   808→            if (!combo) return { ok: false, error: "Selected combination not found." };
   809→            for (const itemId of combo.itemIds) {
   810→              rows.push({
   811→                id: `reg-${classId}-${student.id}-${itemId}`,
   812→                studentId: student.id,
   813→                classId,
   814→                itemId,
   815→                academicYear: year,
   816→                classification: "optional",
   817→                source: "combination",
   818→                combinationId,
   819→                status: "active",
   820→                registeredAt: now,
   821→                droppedAt: null,
   822→              });
   823→            }
   824→          }
   825→        }
   826→        set((s) => ({
   827→          registrations: [
   828→            ...s.registrations.filter((r) => !rows.some((row) => row.id === r.id)),
   829→            ...rows,
   830→          ],
   831→          audit: [
   832→            ...s.audit,
   833→            audit(
   834→              s,
   835→              "bulk-register",
   836→              "student_registration",
   837→              adoptionKey(classId, year),
   838→              `${rows.length} registration rows upserted for ${classId}`,
   839→            ),
   840→          ],
   841→        }));
   842→        return { ok: true, registered: rows.length };
   843→      },
   844→    }),
   845→    {
   846→      name: "josephian-academic-admin",
   847→      skipHydration: true,
   848→      merge: (persisted, current) => {
   849→        const merged = {
   850→          ...current,
   851→          ...(typeof persisted === "object" && persisted ? persisted : {}),
   852→        } as AcademicAdminState;
   853→        if (merged.seedVersion !== ACADEMIC_SEED_VERSION || !Array.isArray(merged.items)) {
   854→          return { ...current, audit: current.audit };
   855→        }
   856→        return merged;
   857→      },
   858→    },
   859→  ),
   860→);

   862→/* Bulk helper: resolve the adopted curriculum + the class's students. */
   863→function bulkHelpers(data: AcademicAdminData, classId: string, year: string) {
   864→  const curriculum = data.curricula.find(
   865→    (c) => c.id === data.adoptions[adoptionKey(classId, year)]?.curriculumId,
   866→  );
   867→  return { adoptedCurriculum: curriculum, studentsInClassOf: studentsInClass(classId) };
   868→}
