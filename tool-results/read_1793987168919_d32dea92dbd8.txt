    26→    26→/* ================================================================
    27→    27→ * Academic Administration — data model (module task §10)
    28→    28→ * ================================================================
    29→    29→ * The app has no server database yet: every module (report templates,
    30→    30→ * nursery assessments, viewer roles) persists its own relational records
    31→    31→ * in a zustand `persist` slice. This module follows the same pattern and
    32→    32→ * implements the logical entities from task §10 as FIRST-CLASS RELATIONAL
    33→    33→ * RECORDS (never comma-separated strings / JSON arrays):
    34→    34→ *
    35→    35→ *   academic_items               → AcademicItem
    36→    36→ *   curricula + curriculum_items → Curriculum (items embedded, versioned)
    37→    37→ *   class_curricula              → ClassCurriculumAdoption (key classId|year)
    38→    38→ *   teacher_item_assignments     → TeacherItemAssignment (history kept)
    39→    39→ *   subject_combinations         → SubjectCombination (+ item ids)
    40→    40→ *   student_subject_registrations→ StudentSubjectRegistration (one row each)
    41→    41→ *   assessment_configurations    → CurriculumItem.assessment + overrides
    42→    42→ *   academic_years/terms         → AcademicYearDef
    43→    43→ *   audit trail                  → AuditEntry
    44→    44→ *
    45→    45→ * SEEDING GUARANTEE: buildSeedAcademicState() derives every record from the
    46→    46→ * existing demo data (subjects, classSubjectsFor, subjectTeacherOf,
    47→    47→ * isEnrolledIn, nursery areas) deterministically. With the seed intact the
    48→    48→ * effective* helpers below return EXACTLY what classSubjectsFor /
    49→    49→ * eligibleStudentsFor / subjectTeacherOf return today, so marks entry,
    50→    50→ * report cards and analytics behave identically until an administrator
    51→    51→ * edits something.
    52→    52→ * ================================================================ */

   212→   212→/** Nursery template area names → catalogue learning areas (registered once). */
   213→   213→function seedLearningAreaItems(): AcademicItem[] {
   214→   214→  const seen = new Map<string, AcademicItem>();
   215→   215→  const usedCodes = new Set<string>([
   216→   216→    ...subjects.map((s) => s.code.toUpperCase()),
   217→   217→  ]);
   218→   218→  const order = [
   219→   219→    "Physical Development",
   220→   220→    "Language Development",
   221→   221→    "Social and Emotional Development",
   222→   222→    "Cognitive Development",
   223→   223→    "Kindergarten Arithmetic",
   224→   224→    "Early Language & Communication",
   225→   225→    "Creative Arts and Sensory Play",
   226→   226→    "Social Skills",
   227→   227→    "Pre-Early Language & Communication",
   228→   228→    "Early Lifeskills (Relations)",
   229→   229→    "Creative Art & Sensory Play",
   230→   230→    "Social & Developmental Learning Skills",
   231→   231→    "Early Numeracy",
   232→   232→    "Sports, Arts & Crafts",
   233→   233→    "Health & Environmental Care",
   234→   234→    "Culture & Religion",
   235→   235→  ];
   236→   236→  for (const areas of EARLY_YEARS_AREA_SOURCES) {
   237→   237→    for (const a of areas) {
   238→   238→      if (seen.has(a.name)) continue;
   239→   239→      let code = laCode(a.name);
   240→   240→      let n = 2;
   241→   241→      while (usedCodes.has(code)) code = `${laCode(a.name)}-${n++}`;
   242→   242→      usedCodes.add(code);
   243→   243→      seen.set(a.name, {
   244→   244→        id: `item-la-${a.name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)/g, "")}`,
   245→   245→        name: a.name,
   246→   246→        code,
   247→   247→        type: "learning-area",
   248→   248→        category: learningAreaCategory(a.name),
   249→   249→        description: `Early Years developmental learning area (curriculum structure from the school's report templates).`,
   250→   250→        sections: ["Early Years"],
   251→   251→        status: "active",
   252→   252→        createdAt: NOW,
   253→   253→        updatedAt: NOW,
   254→   254→      });
   255→   255→    }
   256→   256→  }
   257→   257→  // Keep a stable, human-friendly order in the catalogue table.
   258→   258→  return [...seen.values()].sort(
   259→   259→    (a, b) =>
   260→   260→      order.indexOf(a.name) - order.indexOf(b.name) || a.name.localeCompare(b.name),
   261→   261→  );
   262→   262→}
   263→   263→
   264→   264→function laCode(name: string): string {
   265→   265→  const words = name.replace(/[^A-Za-z ]/g, "").split(/\s+/).filter(Boolean);
   266→   266→  return words.map((w) => w.slice(0, 3).toUpperCase()).join("").slice(0, 6);
   267→   267→}
   268→   268→
   269→   269→function learningAreaCategory(name: string): string {
   270→   270→  const n = name.toLowerCase();
   271→   271→  if (n.includes("numer") || n.includes("arithmetic")) return "Numeracy";
   272→   272→  if (n.includes("language") || n.includes("communication")) return "Communication";
   273→   273→  if (n.includes("physical") || n.includes("sports") || n.includes("health"))
   274→   274→    return "Physical & Health";
   275→   275→  if (n.includes("creativ") || n.includes("art") || n.includes("culture"))
   276→   276→    return "Creative & Culture";
   277→   277→  return "Developmental";
   278→   278→}

   342→   342→
   343→   343→  curricula.push(
   344→   344→    subjectCurriculumForLevel(

   346→   346→      "Primary curriculum — Standard 5",

   349→   349→    ),

   469→   469→      academicYear: year,
   470→   470→      classification,
   471→   471→      source,
   472→   472→      combinationId: null,

   474→   474→      registeredAt: NOW,

   823→   823→export interface RegistrationValidation {
   824→   824→  errors: string[];
   825→   825→  warnings: string[];
   826→   826→}
   827→   827→
   828→   828→export function validateRegistration(
   829→   829→  data: Pick<AcademicAdminData, "curricula" | "adoptions" | "registrations" | "items" | "combinations" | "registrationModes">,
   830→   830→  studentId: string,
   831→   831→  classId: string,
   832→   832→  optionalItemIds: string[],
   833→   833→  combinationId: string | null = null,
   834→   834→  year = "2026",
   835→   835→): RegistrationValidation {
   836→   836→  const errors: string[] = [];
   837→   837→  const warnings: string[] = [];
   838→   838→  const curriculum = adoptedCurriculum(data, classId, year);
   839→   839→  if (!curriculum) {
   840→   840→    errors.push("The class has no adopted curriculum.");
   841→   841→    return { errors, warnings };
   842→   842→  }
   843→   843→  const optionalAvailable = curriculum.items.filter((ci) => ci.classification === "optional");
   844→   844→  const availableIds = new Set(optionalAvailable.map((ci) => ci.itemId));
   845→   845→  const active = new Set(
   846→   846→    data.registrations
   847→   847→      .filter(
   848→   848→        (r) =>
   849→   849→          r.classId === classId &&
   850→   850→          r.studentId === studentId &&
   851→   851→          r.academicYear === year &&
   852→   852→          r.status === "active" &&
   853→   853→          r.classification === "optional",
   854→   854→      )
   855→   855→      .map((r) => r.itemId),
   856→   856→  );
   857→   857→  const activeAny = new Set(
   858→   858→    data.registrations
   859→   859→      .filter(
   860→   860→        (r) =>
   861→   861→          r.classId === classId &&
   862→   862→          r.studentId === studentId &&
   863→   863→          r.academicYear === year &&
   864→   864→          r.status === "active",
   865→   865→      )
   866→   866→      .map((r) => r.itemId),
   867→   867→  );
   868→   868→  for (const id of optionalItemIds) {
   869→   869→    if (!availableIds.has(id))
   870→   870→      errors.push(`${itemById(data, id)?.name ?? id} is not offered by this class's curriculum.`);
   871→   871→    if (activeAny.has(id))
   872→   872→      errors.push(`${itemById(data, id)?.name ?? id} is already registered.`);
   873→   873→  }
   874→   874→  const total = [...active].length + optionalItemIds.filter((id) => !active.has(id)).length;
   875→   875→  const mode = data.registrationModes[adoptionKey(classId, year)] ?? "individual";
   876→   876→  if (combinationId) {
   877→   877→    const combo = data.combinations.find((c) => c.id === combinationId);
   878→   878→    if (!combo) errors.push("Selected combination not found.");
   879→   879→    else {
   880→   880→      if (combo.level !== curriculum.level)
   881→   881→        errors.push(`${combo.name} is offered for ${combo.level}, not ${curriculum.level}.`);
   882→   882→      for (const id of optionalItemIds)
   883→   883→        if (!combo.itemIds.includes(id))
   884→   884→          errors.push(
   885→   885→            `${itemById(data, id)?.name ?? id} is not part of ${combo.name}.`,
   886→   886→          );
   887→   887→    }
   888→   888→  }
   889→   889→  if (mode !== "individual") {
   890→   890→    const combo = data.combinations.find((c) => c.id === combinationId);
   891→   891→    const min = combo?.minSelections ?? null;
   892→   892→    const max = combo?.maxSelections ?? null;
   893→   893→    if (min !== null && total < min)
   894→   894→      errors.push(`At least ${min} optional subjects must be selected.`);
   895→   895→    if (max !== null && total > max)
   896→   896→      errors.push(`At most ${max} optional subjects may be selected.`);
   897→   897→  }
   898→   898→  return { errors, warnings };
   899→   899→}
