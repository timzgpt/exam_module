    26→/* ================================================================
    27→ * Academic Administration — data model (module task §10)
    28→ * ================================================================
    29→ * The app has no server database yet: every module (report templates,
    30→ * nursery assessments, viewer roles) persists its own relational records
    31→ * in a zustand `persist` slice. This module follows the same pattern and
    32→ * implements the logical entities from task §10 as FIRST-CLASS RELATIONAL
    33→ * RECORDS (never comma-separated strings / JSON arrays):
    34→ *
    35→ *   academic_items               → AcademicItem
    36→ *   curricula + curriculum_items → Curriculum (items embedded, versioned)
    37→ *   class_curricula              → ClassCurriculumAdoption (key classId|year)
    38→ *   teacher_item_assignments     → TeacherItemAssignment (history kept)
    39→ *   subject_combinations         → SubjectCombination (+ item ids)
    40→ *   student_subject_registrations→ StudentSubjectRegistration (one row each)
    41→ *   assessment_configurations    → CurriculumItem.assessment + overrides
    42→ *   academic_years/terms         → AcademicYearDef
    43→ *   audit trail                  → AuditEntry
    44→ *
    45→ * SEEDING GUARANTEE: buildSeedAcademicState() derives every record from the
    46→ * existing demo data (subjects, classSubjectsFor, subjectTeacherOf,
    47→ * isEnrolledIn, nursery areas) deterministically. With the seed intact the
    48→ * effective* helpers below return EXACTLY what classSubjectsFor /
    49→ * eligibleStudentsFor / subjectTeacherOf return today, so marks entry,
    50→ * report cards and analytics behave identically until an administrator
    51→ * edits something.
    52→ * ================================================================ */

   212→/** Nursery template area names → catalogue learning areas (registered once). */
   213→function seedLearningAreaItems(): AcademicItem[] {
   214→  const seen = new Map<string, AcademicItem>();
   215→  const usedCodes = new Set<string>([
   216→    ...subjects.map((s) => s.code.toUpperCase()),
   217→  ]);
   218→  const order = [
   219→    "Physical Development",
   220→    "Language Development",
   221→    "Social and Emotional Development",
   222→    "Cognitive Development",
   223→    "Kindergarten Arithmetic",
   224→    "Early Language & Communication",
   225→    "Creative Arts and Sensory Play",
   226→    "Social Skills",
   227→    "Pre-Early Language & Communication",
   228→    "Early Lifeskills (Relations)",
   229→    "Creative Art & Sensory Play",
   230→    "Social & Developmental Learning Skills",
   231→    "Early Numeracy",
   232→    "Sports, Arts & Crafts",
   233→    "Health & Environmental Care",
   234→    "Culture & Religion",
   235→  ];
   236→  for (const areas of EARLY_YEARS_AREA_SOURCES) {
   237→    for (const a of areas) {
   238→      if (seen.has(a.name)) continue;
   239→      let code = laCode(a.name);
   240→      let n = 2;
   241→      while (usedCodes.has(code)) code = `${laCode(a.name)}-${n++}`;
   242→      usedCodes.add(code);
   243→      seen.set(a.name, {
   244→        id: `item-la-${a.name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)/g, "")}`,
   245→        name: a.name,
   246→        code,
   247→        type: "learning-area",
   248→        category: learningAreaCategory(a.name),
   249→        description: `Early Years developmental learning area (curriculum structure from the school's report templates).`,
   250→        sections: ["Early Years"],
   251→        status: "active",
   252→        createdAt: NOW,
   253→        updatedAt: NOW,
   254→      });
   255→    }
   256→  }
   257→  // Keep a stable, human-friendly order in the catalogue table.
   258→  return [...seen.values()].sort(
   259→    (a, b) =>
   260→      order.indexOf(a.name) - order.indexOf(b.name) || a.name.localeCompare(b.name),
   261→  );
   262→}
   263→
   264→function laCode(name: string): string {
   265→  const words = name.replace(/[^A-Za-z ]/g, "").split(/\s+/).filter(Boolean);
   266→  return words.map((w) => w.slice(0, 3).toUpperCase()).join("").slice(0, 6);
   267→}
   268→
   269→function learningAreaCategory(name: string): string {
   270→  const n = name.toLowerCase();
   271→  if (n.includes("numer") || n.includes("arithmetic")) return "Numeracy";
   272→  if (n.includes("language") || n.includes("communication")) return "Communication";
   273→  if (n.includes("physical") || n.includes("sports") || n.includes("health"))
   274→    return "Physical & Health";
   275→  if (n.includes("creativ") || n.includes("art") || n.includes("culture"))
   276→    return "Creative & Culture";
   277→  return "Developmental";
   278→}

   612→ * Effective per-class subject allocation (report-card / marks-grid order)
   613→ * built from the adopted curriculum. Falls back to the demo allocation when
   614→ * the class has no adoption, so existing pipelines never break.
   615→ */
   616→export function effectiveClassSubjectsFor(
   617→  data: Pick<AcademicAdminData, "curricula" | "adoptions" | "assignments" | "assessmentOverrides">,
   618→  classId: string,
   619→  year = "2026",
   620→): ClassSubject[] {
   621→  const curriculum = adoptedCurriculum(data, classId, year);
   622→  if (!curriculum) return classSubjectsFor(classId);
   623→  const cls = classById(classId);
   624→  return curriculum.items
   625→    .slice()
   626→    .sort((a, b) => a.order - b.order)
   627→    .map((ci): ClassSubject => {
   628→      const subject = subjectById(ci.itemId.replace("item-", ""));
   629→      const assignment = activeAssignmentFor(data, classId, ci.itemId);
   630→      const override = data.assessmentOverrides[assessmentKey(classId, ci.itemId)];
   631→      const ca = override?.caMarks ?? ci.assessment.caMarks;
   632→      const exam = override?.examMarks ?? ci.assessment.examMarks;
   633→      return {
   634→        subjectId: ci.itemId.replace("item-", ""),
   635→        classId,
   636→        type: ci.classification === "core" ? "core" : "subsidiary",
   637→        chartLabel: ci.chartLabel || subject?.code || "",
   638→        teacherId:
   639→          assignment?.teacherId ??
   640→          subjectTeacherOf(ci.itemId.replace("item-", ""), classId)?.id ??
   641→          cls?.classTeacherId ??
   642→          teacherById("")?.id ??
   643→          "",
   644→        periodsPerWeek: ci.periodsPerWeek,
   645→        examPapers: ci.examPapers,
   646→        caMarks: ca,
   647→        examMarks: exam,
   648→        fullMarks: override?.fullMarks ?? (ca + exam || ci.assessment.fullMarks),
   649→      };
   650→    });
   651→}

   654→export function activeAssignmentFor(
   655→  data: Pick<AcademicAdminData, "assignments">,
   656→  classId: string,
   657→  itemId: string,
   658→  year = "2026",
   659→): TeacherItemAssignment | undefined {

   662→      a.classId === classId &&
   663→      a.itemId === itemId &&

   671→/**
   672→ * Subject ids the viewer may enter marks for in a class (task §12/§13).
   673→ * When the academic store holds teacher assignments for the class they are
   674→ * authoritative and the scoping is strict: a subject teacher with no active
   675→ * assignment rows in a configured class sees NOTHING there. Classes without
   676→ * any assignment rows fall back to the legacy demo derivation so the app
   677→ * keeps working if configuration is reset.
   678→ */
   679→export function effectiveAssignedSubjectIds(
   680→  data: Pick<AcademicAdminData, "curricula" | "adoptions" | "assignments" | "assessmentOverrides">,
   681→  viewer: { role: string; teacherId: string | null },
   682→  classId: string,
   683→): string[] {
   684→  const all = effectiveClassSubjectsFor(data, classId).map((cs) => cs.subjectId);
   685→  if (viewer.role !== "subject-teacher" || !viewer.teacherId) return all;
   686→  const classConfigured = data.assignments.some(
   687→    (a) => a.classId === classId && a.academicYear === "2026",
   688→  );
   689→  if (!classConfigured) {
   690→    return all.filter((sid) => subjectTeacherOf(sid, classId)?.id === viewer.teacherId);
   691→  }
   692→  const mine = data.assignments.filter(
   693→    (a) =>
   694→      a.classId === classId &&
   695→      a.teacherId === viewer.teacherId &&
   696→      a.status === "assigned" &&
   697→      a.endsAt === null &&
   698→      a.academicYear === "2026",
   699→  );
   700→  const mineItems = new Set(mine.map((a) => a.itemId));
   701→  return all.filter((sid) => mineItems.has(`item-${sid}`));
   702→}

   776→/** Teacher workload: distinct classes and items from active assignments. */
   777→export function workloadFor(
   778→  data: Pick<AcademicAdminData, "assignments" | "curricula">,
   779→  teacherId: string,
   780→  year = "2026",
   781→): { classes: string[]; items: { itemId: string; classId: string }[]; periods: number } {
   782→  const rows = data.assignments.filter(
   783→    (a) => a.teacherId === teacherId && a.status === "assigned" && a.endsAt === null && a.academicYear === year,
   784→  );
   785→  const items = rows.map((a) => ({ itemId: a.itemId, classId: a.classId }));
   786→  let periods = 0;
   787→  for (const r of rows) {
   788→    const curriculum = data.curricula.find((c) =>
   789→      c.items.some((ci) => ci.itemId === r.itemId),
   790→    );
   791→    periods += curriculum?.items.find((ci) => ci.itemId === r.itemId)?.periodsPerWeek ?? 0;
   792→  }
   793→  return { classes: [...new Set(rows.map((a) => a.classId))], items, periods };
   794→}

   798→export function validateItem(
   799→  data: AcademicAdminData,
   800→  draft: Pick<AcademicItem, "name" | "code" | "type" | "sections" | "category">,
   801→  editingId?: string,
   802→): string[] {
   803→  const errors: string[] = [];
   804→  if (!draft.name.trim()) errors.push("Name is required.");
   805→  if (!draft.code.trim()) errors.push("Code is required.");
   806→  if (!draft.category?.trim()) errors.push("Category is required.");
   807→  if (draft.sections.length === 0)
   808→    errors.push("Choose at least one applicable school section.");
   809→  const code = draft.code.trim().toUpperCase();
   810→  const clash = data.items.find(
   811→    (i) => i.code.toUpperCase() === code && i.id !== editingId,
   812→  );
   813→  if (clash) errors.push(`Code ${code} is already used by ${clash.name}.`);
   814→  const nameClash = data.items.find(
   815→    (i) => i.name.trim().toLowerCase() === draft.name.trim().toLowerCase() && i.id !== editingId,
   816→  );
   817→  if (nameClash) errors.push(`An item named "${nameClash.name}" already exists.`);
   818→  return errors;
   819→}

   821→/* ── Registration validation (task §5.C Step 4 / §8) ──────────────── */
   822→
   823→export interface RegistrationValidation {
   824→  errors: string[];
   825→  warnings: string[];
   826→}
   827→
   828→export function validateRegistration(
   829→  data: Pick<AcademicAdminData, "curricula" | "adoptions" | "registrations" | "items" | "combinations" | "registrationModes">,
   830→  studentId: string,
   831→  classId: string,
   832→  optionalItemIds: string[],
   833→  combinationId: string | null = null,
   834→  year = "2026",
   835→): RegistrationValidation {
   836→  const errors: string[] = [];
   837→  const warnings: string[] = [];
   838→  const curriculum = adoptedCurriculum(data, classId, year);
   839→  if (!curriculum) {
   840→    errors.push("The class has no adopted curriculum.");
   841→    return { errors, warnings };
   842→  }
   843→  const optionalAvailable = curriculum.items.filter((ci) => ci.classification === "optional");
   844→  const availableIds = new Set(optionalAvailable.map((ci) => ci.itemId));
   845→  const active = new Set(
   846→    data.registrations
   847→      .filter(
   848→        (r) =>
   849→          r.classId === classId &&
   850→          r.studentId === studentId &&
   851→          r.academicYear === year &&
   852→          r.status === "active" &&
   853→          r.classification === "optional",
   854→      )
   855→      .map((r) => r.itemId),
   856→  );
   857→  const activeAny = new Set(
   858→    data.registrations
   859→      .filter(
   860→        (r) =>
   861→          r.classId === classId &&
   862→          r.studentId === studentId &&
   863→          r.academicYear === year &&
   864→          r.status === "active",
   865→      )
   866→      .map((r) => r.itemId),
   867→  );
   868→  for (const id of optionalItemIds) {
   869→    if (!availableIds.has(id))
   870→      errors.push(`${itemById(data, id)?.name ?? id} is not offered by this class's curriculum.`);
   871→    if (activeAny.has(id))
   872→      errors.push(`${itemById(data, id)?.name ?? id} is already registered.`);
   873→  }
   874→  const total = [...active].length + optionalItemIds.filter((id) => !active.has(id)).length;
   875→  const mode = data.registrationModes[adoptionKey(classId, year)] ?? "individual";
   876→  if (combinationId) {
   877→    const combo = data.combinations.find((c) => c.id === combinationId);
   878→    if (!combo) errors.push("Selected combination not found.");
   879→    else {
   880→      if (combo.level !== curriculum.level)
   881→        errors.push(`${combo.name} is offered for ${combo.level}, not ${curriculum.level}.`);
   882→      for (const id of optionalItemIds)
   883→        if (!combo.itemIds.includes(id))
   884→          errors.push(
   885→            `${itemById(data, id)?.name ?? id} is not part of ${combo.name}.`,
   886→          );
   887→    }
   888→  }
   889→  if (mode !== "individual") {
   890→    const combo = data.combinations.find((c) => c.id === combinationId);
   891→    const min = combo?.minSelections ?? null;
   892→    const max = combo?.maxSelections ?? null;
   893→    if (min !== null && total < min)
   894→      errors.push(`At least ${min} optional subjects must be selected.`);
   895→    if (max !== null && total > max)
   896→      errors.push(`At most ${max} optional subjects may be selected.`);
   897→  }
   898→  return { errors, warnings };
   899→}

   901→/* ── Class configuration completeness (task §2 / §13) ─────────────── */
   902→
   903→export interface ClassCompleteness {
   904→  allocation: boolean;
   905→  teachers: boolean;
   906→  registrations: boolean;
   907→  assessment: boolean;
   908→  score: number; // 0-100
   909→}
   910→
   911→export function classCompleteness(
   912→  data: Pick<AcademicAdminData, "curricula" | "adoptions" | "assignments" | "registrations" | "items">,
   913→  classId: string,
   914→  year = "2026",
   915→): ClassCompleteness {
   916→  const curriculum = adoptedCurriculum(data, classId, year);
   917→  const allocation = !!curriculum;
   918→  const items = curriculum?.items ?? [];
   919→  const teachers =
   920→    items.length > 0 &&
   921→    items.every((ci) => activeAssignmentFor(data, classId, ci.itemId, year));
   922→  const students =
   923→    classById(classId) && !nurseryClasses.some((c) => c.id === classId)
   924→      ? studentsInClass(classId)
   925→      : nurseryStudents.filter((s) => s.classId === classId);
   926→  const registrations =
   927→    items.length > 0 &&
   928→    students.length > 0 &&
   929→    students.every((s) => {
   930→      const rows = data.registrations.filter(
   931→        (r) =>
   932→          r.classId === classId && r.studentId === s.id && r.academicYear === year && r.status === "active",
   933→      );
   934→      return rows.length > 0;
   935→    });
   936→  const assessment =
   937→    items.length > 0 &&
   938→    items.every(
   939→      (ci) =>
   940→        ci.assessment.developmentalModel !== null ||
   941→        ci.assessment.caMarks + ci.assessment.examMarks > 0,
   942→    );
   943→  const flags = [allocation, teachers, registrations, assessment];
   944→  const score = Math.round((flags.filter(Boolean).length / flags.length) * 100);
   945→  return { allocation, teachers, registrations, assessment, score };
   946→}
