    26→function hashStr(s: string): number {
    27→  let h = 2166136261;
    28→  for (let i = 0; i < s.length; i++) {
    29→    h ^= s.charCodeAt(i);
    30→    h = Math.imul(h, 16777619);
    31→  }
    32→  return h >>> 0;
    33→}
    34→
    35→function mulberry32(seed: number): () => number {
    36→  let a = seed;
    37→  return () => {
    38→    a |= 0;
    39→    a = (a + 0x6d2b79f5) | 0;
    40→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    41→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    42→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    43→  };
    44→}

   111→function buildStudents(): Student[] {
   112→  const list: Student[] = [];
   113→  for (const cls of classes) {
   114→    const short = CLASS_SHORT[cls.id];
   115→    for (let i = 0; i < 12; i++) {
   116→      const seed = hashStr(cls.id + i);
   117→      const gender: "M" | "F" = seed % 2 === 0 ? "F" : "M";
   118→      const pool = gender === "F" ? FEMALE_NAMES : MALE_NAMES;
   119→      const first = pool[(seed >>> 3) % pool.length];
   120→      const middle = MIDDLE_NAMES[(seed >>> 5) % MIDDLE_NAMES.length];
   121→      const surname = SURNAMES[(seed >>> 7) % SURNAMES.length];
   122→      const id = cls.id === DEMO_IDS.classId && i === 0 ? DEMO_IDS.studentId : `${short}-${String(i + 1).padStart(2, "0")}`;

   305→/** Pairs with missing entries so deadline tracker / readiness look real. */
   306→const NULL_PATCHES: Record<string, string[] | "ALL"> = {
   307→  "cls-4a|s5": ["f4a-03", "f4a-07", "f4a-10", "f4a-11"],
   308→  "cls-2a|s7": "ALL",
   309→  "cls-1b|s9": ["f1b-05", "f1b-09"],

   311→
   312→const marksCache = new Map<string, Record<string, number | null>>();
   313→
   314→export function getMarks(
   315→  classId: string,
   316→  subjectId: string,
   317→): Record<string, number | null> {
   318→  const key = `${classId}|${subjectId}`;
   319→  const cached = marksCache.get(key);
   320→  if (cached) return cached;
   321→
   322→  const rnd = mulberry32(hashStr(key));
   323→  const classOffset = (hashStr(classId) % 13) - 6;
   324→  const subjectOffset = (hashStr(subjectId) % 15) - 7;
   325→  const result: Record<string, number | null> = {};
   326→  for (const s of studentsInClass(classId)) {
   327→    const ability = 44 + (hashStr(s.id) % 46); // 44-89 latent ability
   328→    const jitter = rnd() * 18 - 9;
   329→    const mark = Math.round(ability + classOffset + subjectOffset + jitter);
   330→    result[s.id] = Math.max(14, Math.min(99, mark));
   331→  }
   332→  const patch = NULL_PATCHES[key];
   333→  if (patch === "ALL") {
   334→    for (const sid of Object.keys(result)) result[sid] = null;
   335→  } else if (patch) {
   336→    for (const sid of patch) result[sid] = null;
   337→  }
   338→  marksCache.set(key, result);
   339→  return result;
   340→}

   393→export function commentFor(classId: string, studentId: string): CommentRecord {
   394→  const rnd = mulberry32(hashStr(`comment|${classId}|${studentId}`));
   395→  const missing = COMMENTS_MISSING.has(studentId);
   396→  const draft = COMMENTS_DRAFT.has(studentId);
   397→  const idx = Math.floor(rnd() * commentTemplates.length);
   398→  const second = Math.floor(rnd() * commentTemplates.length);
   399→  return {
   400→    studentId,
   401→    classTeacher: missing ? null : commentTemplates[idx],
   402→    headTeacher: missing || rnd() > 0.75 ? null : commentTemplates[second],
   403→    status: missing ? "draft" : draft ? "draft" : "submitted",
   404→    updatedAt: `2026-03-0${1 + Math.floor(rnd() * 8)}`,
   405→  };

   658→export type VerifyOutcome =
   659→  | { state: "valid"; record: VerificationRecord }
   660→  | { state: "not-found" }
   661→  | { state: "tampered"; record: VerificationRecord }
   662→  | { state: "revoked"; record: VerificationRecord };

   664→export function verifyCode(raw: string): VerifyOutcome {
   665→  const code = raw.trim().toUpperCase();
   666→  if (code === "JS26-T1-0147-0000") {
   667→    return {
   668→      state: "tampered",
   669→      record: verificationRecords[0],
   670→    };
   671→  }
   672→  const found = verificationRecords.find((r) => r.code === code);
   673→  if (!found) return { state: "not-found" };
   674→  if (found.revoked) return { state: "revoked", record: found };
   675→  return { state: "valid", record: found };
   676→}

   682→export interface StudentResult {
   683→  student: Student;
   684→  marks: Record<string, number | null>;
   685→  total: number;
   686→  count: number;
   687→  average: number;
   688→  grade: string;
   689→  rank: number;
   690→}

   692→const classResultsCache = new Map<string, StudentResult[]>();
   693→
   694→export function classResults(classId: string): StudentResult[] {
   695→  const cached = classResultsCache.get(classId);
   696→  if (cached) return cached;
   697→  const rows = studentsInClass(classId).map((student) => {
   698→    const marks: Record<string, number | null> = {};
   699→    let total = 0;
   700→    let count = 0;
   701→    for (const sub of subjects) {
   702→      const m = getMarks(classId, sub.id)[student.id];
   703→      marks[sub.id] = m;
   704→      if (m !== null) {
   705→        total += m;
   706→        count += 1;
   707→      }
   708→    }
   709→    const average = count > 0 ? total / count : 0;
   710→    return { student, marks, total, count, average, grade: count > 0 ? gradeFor(average, defaultScheme)?.grade ?? "F" : "-", rank: 0 };
   711→  });
   712→  rows.sort((a, b) => b.average - a.average);
   713→  rows.forEach((r, i) => (r.rank = i + 1));
   714→  classResultsCache.set(classId, rows);
   715→  return rows;
   716→}

   720→  if (!student) return undefined;

   754→export interface CycleStats {
   755→  totalStudents: number;
   756→  average: number;
   757→  passRate: number;
   758→  gradeDistribution: Record<string, number>;
   759→  classAverages: { classId: string; name: string; avg: number }[];
   760→  subjectAverages: { subjectId: string; name: string; department: string; avg: number; passRate: number }[];
   761→  marksCompletion: number;
   762→  commentsCompletion: number;
   763→  topStudents: { name: string; className: string; avg: number }[];
   764→}
   765→
   766→const cycleStatsCache = new Map<string, CycleStats>();
   767→
   768→export function cycleStatsFor(cycleId: string): CycleStats {
   769→  const cached = cycleStatsCache.get(cycleId);
   770→  if (cached) return cached;
   771→  const all = classes.flatMap((c) => classResults(c.id));
   772→  const allMarks = all.flatMap((r) => Object.values(r.marks).filter((v): v is number => v !== null));
   773→  const average = allMarks.length ? allMarks.reduce((a, b) => a + b, 0) / allMarks.length : 0;
   774→  const gradeDistribution: Record<string, number> = {};
   775→  for (const b of boundaries) gradeDistribution[b.grade] = 0;
   776→  for (const m of allMarks) {
   777→    const g = gradeFor(m, defaultScheme)?.grade;
   778→    if (g) gradeDistribution[g] += 1;
   779→  }
   780→  const totalEntries = marksDeadlines.reduce((a, d) => a + d.entriesTotal, 0);
   781→  const doneEntries = marksDeadlines.reduce((a, d) => a + d.entriesDone, 0);
   782→  const commentTotals = classes.reduce(
   783→    (acc, c) => {
   784→      const p = commentProgress(c.id);
   785→      return { done: acc.done + p.done, total: acc.total + p.total };
   786→    },
   787→    { done: 0, total: 0 },
   788→  );
   789→  const stats: CycleStats = {
   790→    totalStudents: all.length,
   791→    average,
   792→    passRate: allMarks.length ? Math.round((allMarks.filter((m) => m >= defaultScheme.passMark).length / allMarks.length) * 100) : 0,
   793→    gradeDistribution,
   794→    classAverages: classes.map((c) => ({
   795→      classId: c.id,
   796→      name: c.name,
   797→      avg: classResults(c.id).reduce((a, r) => a + r.average, 0) / classResults(c.id).length,
   798→    })),
   799→    subjectAverages: subjects.map((sub) => {
   800→      const ms = classes.flatMap((c) =>
   801→        Object.values(getMarks(c.id, sub.id)).filter((v): v is number => v !== null),
   802→      );
   803→      return {
   804→        subjectId: sub.id,
   805→        name: sub.name,
   806→        department: sub.department,
   807→        avg: ms.length ? ms.reduce((a, b) => a + b, 0) / ms.length : 0,
   808→        passRate: ms.length ? Math.round((ms.filter((m) => m >= defaultScheme.passMark).length / ms.length) * 100) : 0,
   809→      };
   810→    }),
   811→    marksCompletion: totalEntries ? Math.round((doneEntries / totalEntries) * 100) : 0,
   812→    commentsCompletion: commentTotals.total ? Math.round((commentTotals.done / commentTotals.total) * 100) : 0,
   813→    topStudents: all
   814→      .slice()
   815→      .sort((a, b) => b.average - a.average)
   816→      .slice(0, 5)
   817→      .map((r) => ({ name: r.student.name, className: classNameOf(r.student.classId), avg: r.average })),
   818→  };
   819→  cycleStatsCache.set(cycleId, stats);
   820→  return stats;
   821→}

   823→export interface DepartmentStat {
   824→  department: string;
   825→  hodName: string;
   826→  avg: number;
   827→  passRate: number;
   828→  submissionRate: number;
   829→  subjects: { subjectId: string; name: string; avg: number; passRate: number }[];
   830→  teachers: { name: string; subject: string; done: number; total: number }[];
   831→}
   832→
   833→export function departmentStats(): DepartmentStat[] {
   834→  const deptNames = [...new Set(subjects.map((s) => s.department))];
   835→  return deptNames.map((dept) => {
   836→    const deptSubjects = subjects.filter((s) => s.department === dept);
   837→    const rows = deptSubjects.map((sub) => {
   838→      const ms = classes.flatMap((c) =>
   839→        Object.values(getMarks(c.id, sub.id)).filter((v): v is number => v !== null),
   840→      );
   841→      return {
   842→        subjectId: sub.id,
   843→        name: sub.name,
   844→        avg: ms.length ? ms.reduce((a, b) => a + b, 0) / ms.length : 0,
   845→        passRate: ms.length ? Math.round((ms.filter((m) => m >= defaultScheme.passMark).length / ms.length) * 100) : 0,
   846→      };
   847→    });
   848→    const teacherRows = deptSubjects.flatMap((sub) => {
   849→      const t = teachers.find((t) => t.subjectIds.includes(sub.id));
   850→      if (!t) return [];
   851→      const rel = marksDeadlines.filter((d) => d.subjectId === sub.id);
   852→      const done = rel.reduce((a, d) => a + d.entriesDone, 0);
   853→      const total = rel.reduce((a, d) => a + d.entriesTotal, 0);
   854→      return [{ name: t.name, subject: sub.name, done, total }];
   855→    });
   856→    const submissionTotal = teacherRows.reduce((a, r) => a + r.total, 0);
   857→    const submissionDone = teacherRows.reduce((a, r) => a + r.done, 0);
   858→    const hod = teachers.find((t) => t.role.includes(`HOD ${dept}`));
   859→    return {
   860→      department: dept,
   861→      hodName: hod?.name ?? "-",
   862→      avg: rows.reduce((a, r) => a + r.avg, 0) / rows.length,
   863→      passRate: Math.round(rows.reduce((a, r) => a + r.passRate, 0) / rows.length),
   864→      submissionRate: submissionTotal ? Math.round((submissionDone / submissionTotal) * 100) : 0,
   865→      subjects: rows,
   866→      teachers: teacherRows,
   867→    };
   868→  });
   869→}
