    54→/**
    55→ * The card is printed as pure white paper laid out entirely with bordered
    56→ * tables (the school's requested "totally white, only table" design). The
    57→ * single accent colour is the brand maroon #5C0A14 — the exact maroon of
    58→ * the Josephian Schools shield — used for the title band and every table
    59→ * header row; all other fills are white.
    60→ */
    61→const BRAND = "#5C0A14";
    62→
    63→/* Shared chrome for the card's tables. */
    64→const TBL = "w-full table-fixed border-collapse";
    65→/** Full-width maroon section bar (first row of each section table). */
    66→const SECTION_BAR =
    67→  "border border-border bg-[#5C0A14] px-3 py-1.5 text-center text-[11px] font-bold uppercase tracking-[0.22em] text-white";
    68→/** Maroon column-heading cell. */
    69→const COL_HEAD =
    70→  "border border-border bg-[#5C0A14] px-2 py-1.5 text-[11px] font-bold uppercase tracking-[0.1em] text-white";
    71→/** Standard bordered body cell (white). */
    72→const CELL = "border border-border bg-white px-2 py-1.5 align-middle";
    73→/** Tiny uppercase label used inside bordered cells. */
    74→const CELL_LABEL =
    75→  "text-3xs font-semibold uppercase tracking-[0.12em] text-muted-foreground";
    76→
    77→interface ChartRow {
    78→  name: string;
    79→  achieved: number;
    80→  classAvg: number;
    81→  classTop: number;
    82→}
    83→
    84→interface PerformanceChartSectionProps {
    85→  /** Which report engine printed this card (mid-term vs terminal rules). */
    86→  variant: ChartVariant;
    87→  /** Static data-URI chart image; null renders the fallback paragraph. */
    88→  chartImage: string | null;
    89→  rows: ChartRow[];
    90→  strongest: { name: string; score: number } | null;
    91→  /** Mid-Term primary cards only (chart prompt §8). */
    92→  bestImprovement: { name: string; value: number } | null;
    93→  isPrimary: boolean;
    94→}
    95→
    96→/**
    97→ * "Performance Charts" section (A4 section 6 / chart prompt §8) — also laid
    98→ * out as a table: a maroon section bar and a single bordered body cell
    99→ * holding the badge lines, the static grouped bar chart image, the Terminal
   100→ * HTML legend and a visually hidden data table backing the chart for screen
   101→ * readers (B0: charts include a table fallback).
   102→ */
   103→function PerformanceChartSection({
   104→  variant,
   105→  chartImage,
   106→  rows,
   107→  strongest,
   108→  bestImprovement,
   109→  isPrimary,
   110→}: PerformanceChartSectionProps) {
   111→  // Section heading: "Performance Chart" on secondary terminal cards only.
   112→  const heading = variant === "mid-term" || isPrimary ? "Performance Charts" : "Performance Chart";
   113→  return (
   114→    <table className={cn(TBL, "mt-4")}>
   115→      <tbody>
   116→        <tr>
   117→          <th scope="colgroup" className={SECTION_BAR}>
   118→            {heading}
   119→          </th>
   120→        </tr>
   121→        <tr>
   122→          <td className={CELL}>
   123→            {strongest || bestImprovement ? (
   124→              <div className="mb-1.5 flex flex-wrap gap-x-4 gap-y-0.5 text-2xs font-semibold">
   125→                {strongest ? (
   126→                  <span style={{ color: BRAND }}>
   127→                    ★ Strongest Subject: {strongest.name} — {fmtTotalMark(strongest.score)} / 100
   128→                  </span>
   129→                ) : null}
   130→                {bestImprovement ? (
   131→                  <span className="text-[#2D6A4E]">
   132→                    ▲ Best Improvement: {bestImprovement.name} — +{bestImprovement.value}
   133→                  </span>
   134→                ) : null}
   135→              </div>
   136→            ) : null}
   137→            {chartImage ? (
   138→              <img
   139→                src={chartImage}
   140→                alt="Grouped bar chart comparing achieved marks with the class average and class topper for every assessed subject"
   141→                className="chart-img w-full"
   142→              />
   143→            ) : (
   144→              <p className="text-2xs text-muted-foreground">{CHART_FALLBACK_TEXT}</p>
   145→            )}
   146→            {variant === "terminal" ? (
   147→              // Terminal legend is rendered in HTML, not in the chart image (§9).
   148→              <div className="mt-1 flex flex-wrap items-center justify-center gap-x-4 gap-y-1 text-2xs text-muted-foreground">
   149→                <span className="inline-flex items-center gap-1">
   150→                  <span className="size-2 rounded-sm" style={{ background: "#E74C3C" }} /> Achieved Marks
   151→                </span>
   152→                <span className="inline-flex items-center gap-1">
   153→                  <span className="size-2 rounded-sm" style={{ background: "#3498DB" }} /> Class Avg
   154→                </span>
   155→                <span className="inline-flex items-center gap-1">
   156→                  <span className="size-2 rounded-sm" style={{ background: "#2ECC71" }} /> Class Topper
   157→                </span>
   158→              </div>
   159→            ) : null}
   160→            {/* Chart data-table fallback (B0: charts include a table fallback). */}
   161→            <table className="sr-only">
   162→              <caption>Subject totals against the class average and class top</caption>
   163→              <thead>
   164→                <tr>
   165→                  <th scope="col">Subject</th>
   166→                  <th scope="col">Achieved</th>
   167→                  <th scope="col">Class average</th>
   168→                  <th scope="col">Class top</th>
   169→                </tr>
   170→              </thead>
   171→              <tbody>
   172→                {rows.map((row) => (
   173→                  <tr key={row.name}>
   174→                    <th scope="row">{row.name}</th>
   175→                    <td>{fmtTotalMark(row.achieved)}</td>
   176→                    <td>{fmtTotalMark(row.classAvg)}</td>
   177→                    <td>{fmtTotalMark(row.classTop)}</td>
   178→                  </tr>
   179→                ))}
   180→              </tbody>
   181→            </table>
   182→          </td>
   183→        </tr>
   184→      </tbody>
   185→    </table>
   186→  );
   187→}

   189→export default function ReportCardPage() {
   190→  const params = useParams<{ studentId?: string }>();
   191→  const router = useRouter();
   192→  const { cycle } = useCycleContext();
   193→  const student = studentById(params.studentId ?? "") ?? studentById(DEMO_IDS.studentId)!;
   194→
   195→  const result = studentResultOf(student.id);
   196→  const classmates = studentsInClass(student.classId);
   197→  const classSize = classmates.length;
   198→  const attendance = attendanceRecordFor(student.id);
   199→  const comment = commentFor(student.classId, student.id);
   200→  const classTeacherName =
   201→    teacherById(classById(student.classId)?.classTeacherId)?.name ?? "\u2014";
   202→
   203→  const classSubjects = classSubjectsFor(student.classId);
   204→  const scheme = schemeForClass(student.classId);
   205→  const isPrimary = isPrimaryClass(student.classId);
   206→  // Assessment split shown in the table header (25 % monthly / 75 % terminal).
   207→  const caPct = classSubjects[0]?.caMarks ?? 25;
   208→  const examPct = classSubjects[0]?.examMarks ?? 75;
   209→  // Top band of the class scheme — rendered as the filled grade badge.
   210→  const topGrade = scheme.boundaries[0]?.grade;
   211→  const allRows = classSubjects.map((cs) => {
   212→    const subject = subjectById(cs.subjectId)!;
   213→    const comp = getComponents(student.classId, cs.subjectId)[student.id];
   214→    const enrolled = comp?.enrolled ?? true;
   215→    const mark = enrolled ? result?.marks[cs.subjectId] ?? null : null;
   216→    const grade = mark !== null ? gradeFor(mark, scheme) : undefined;
   217→    return { subject, enrolled, mark, grade, ca: comp?.ca ?? null, exam: comp?.exam ?? null };
   218→  });
   219→  // A subject with no mark at all (both components ABS/NA/NR/PENDING) is
   220→  // omitted from the card entirely; non-enrolled subsidiaries stay with "—".
   221→  // Remaining rows renumber continuously after any omitted subject (the #
   222→  // column below).
   223→  const rows = allRows.filter((r) => !r.enrolled || r.mark !== null);
   224→  const assessed = allRows.filter((r) => r.mark !== null);
   225→  const pointsTotal = assessed.reduce((a, r) => a + (r.grade?.points ?? 0), 0);
   226→  // Secondary division: best 7 subject points; null when fewer than 7 graded
   227→  // (and always null for primary classes — divisions are a NECTA outcome).
   228→  const division = isPrimary ? null : divisionFor(assessed.map((r) => r.grade?.points ?? null));
   229→  const passed = subjectsPassed(assessed.map((r) => r.grade?.grade ?? null));
   230→  // Strongest subject: highest total; ties go to the first in subject order.
   231→  const strongest =
   232→    assessed.length > 0
   233→      ? assessed.reduce((best, r) => ((r.mark as number) > (best.mark as number) ? r : best))
   234→      : undefined;
   235→  // Points and division are hidden on primary cards (Primary A–E bands carry
   236→  // no points) and on any secondary class whose teacher set the Report
   237→  // Overview "No Division" flag to Yes.
   238→  const noDivision = !isPrimary && (classById(student.classId)?.noDivision ?? false);
   239→  const showPoints = !isPrimary && !noDivision;
   240→  // Most improved (primary only): largest CA delta C2 − C1 per subject; a
   241→  // single recorded term counts as 0. Ties go to the first subject in order.
   242→  const mostImproved = (() => {
   243→    if (!isPrimary) return null;
   244→    const deltas = classSubjects.map((cs) => {
   245→      const comp = getComponents(student.classId, cs.subjectId)[student.id];
   246→      const cur = comp?.enrolled ? comp.ca ?? null : null;
   247→      const prev = prevCaOf(student.classId, cs.subjectId)[student.id] ?? null;
   248→      return { subjectId: cs.subjectId, delta: cur !== null && prev !== null ? cur - prev : 0 };
   249→    });
   250→    const best = mostImprovedSubject(deltas);
   251→    return best && best.delta > 0 ? best : null;
   252→  })();
   253→  const mostImprovedSubjectRow = mostImproved
   254→    ? rows.find((r) => r.subject.id === mostImproved.subjectId)
   255→    : undefined;
   256→
   257→  // Chart data (A4 section 6 / chart prompt §3–§4): achieved total vs class
   258→  // average and class top per assessed subject, over the student's enrolled
   259→  // classmates. Class stats follow the cycle's engine: mid-term counts a
   260→  // student with a single assessment at 100%; terminal only counts students
   261→  // with BOTH components (chart prompt §4/§9).
   262→  const variant: ChartVariant = cycle.kind === "mid-term" ? "mid-term" : "terminal";
   263→  const chartRows: ChartRow[] = assessed.map((r) => {
   264→    const cs = classSubjects.find((c) => c.subjectId === r.subject.id);
   265→    const comps = getComponents(student.classId, r.subject.id);
   266→    const scores = Object.values(comps)
   267→      .filter((c) => c.enrolled)
   268→      .map((c) => classScoreFor(variant, c.ca, c.exam, cs?.caMarks ?? 25, cs?.examMarks ?? 75))
   269→      .filter((t): t is number => t !== null);
   270→    return {
   271→      name: r.subject.name,
   272→      achieved: r.mark ?? 0,
   273→      classAvg: scores.length ? scores.reduce((a, b) => a + b, 0) / scores.length : 0,
   274→      classTop: scores.length ? Math.max(...scores) : 0,
   275→    };
   276→  });
   277→  // Series + static chart image (chart prompt §7 call sequence).
   278→  const chartSeries = buildChartSeries(
   279→    chartRows.map((r) => ({ name: r.name, achieved: r.achieved })),
   280→    chartRows.map((r) => r.classAvg),
   281→    chartRows.map((r) => r.classTop),
   282→    strongest?.subject.name ?? "",
   283→    variant,
   284→  );
   285→  const chartImage = generateChartImage(chartSeries, variant);
   286→  // Best-Improvement badge: Mid-Term primary cards only (chart prompt §8).
   287→  const chartBestImprovement =
   288→    variant === "mid-term" && mostImproved && mostImprovedSubjectRow
   289→      ? { name: mostImprovedSubjectRow.subject.name, value: mostImproved.delta }
   290→      : null;
   291→
   292→  const switchStudent = (id: string) => {
   293→    router.push(`/academic/report-cycles/${cycle.id}/students/${id}/report-card`);
   294→  };
   295→
   296→  const downloadPdf = usePrintDownload(
   297→    "Use 'Save as PDF' in the print dialog to download the report card.",
   298→  );
   299→
   300→  // Performance summary cells; the points/division pair appears only where
   301→  // points are printed (secondary classes without the No-Division flag).
   302→  const summaryCells = [
   303→    { label: "Total marks", value: result ? fmtTotalMark(result.total) : "-", accent: false },
   304→    { label: "Average", value: result ? result.average.toFixed(1) : "-", accent: false },
   305→    {
   306→      label: "Overall grade",
   307→      value: result && result.count > 0 ? result.grade : "-",
   308→      accent: true,
   309→    },
   310→    ...(showPoints
   311→      ? [
   312→          {
   313→            label: "Points total",
   314→            value: assessed.length > 0 ? String(pointsTotal) : "-",
   315→            accent: false,
   316→          },
   317→          {
   318→            label: "Division",
   319→            value: division ? `Div ${division.division}` : "—",
   320→            accent: true,
   321→          },
   322→        ]
   323→      : []),
   324→    {
   325→      label: "Subjects passed",
   326→      value: assessed.length > 0 ? `${passed} of ${assessed.length}` : "-",
   327→      accent: false,
   328→    },
   329→    { label: "Class size", value: String(classSize), accent: false },
   330→  ];
   331→
   332→  // Attendance row cells; "Not recorded" appears only when > 0 (A3.8).
   333→  const rateText = attendance.rate === null ? "–" : `${attendance.rate.toFixed(1)}%`;
   334→  const attRatingText =
   335→    attendance.rating === "none" ? null : ATTENDANCE_RATING_TEXT[attendance.rating];
   336→  const attCells = [
   337→    {
   338→      label: "Days attended / recorded",
   339→      value: `${attendance.daysAttended} / ${attendance.daysRecorded}`,
   340→    },
   341→    { label: "Attendance rate", value: rateText },
   342→    { label: "Late", value: String(attendance.late) },
   343→    { label: "Absent", value: String(attendance.absent) },
   344→    { label: "Excused", value: String(attendance.excused) },
   345→    ...(attendance.notRecorded > 0
   346→      ? [
   347→          {
   348→            label: "Not recorded",
   349→            value: `${attendance.notRecorded} day${attendance.notRecorded === 1 ? "" : "s"}`,
   350→          },
   351→        ]
   352→      : []),
   353→  ];
   354→
   355→  return (
   356→    <div className="space-y-5">
   357→      <PageHeader
   358→        title="Report card"
   359→        description={`${student.name} · ${classNameOf(student.classId)} · ${cycle.name}`}
   360→      />
   361→
   362→      {/* Toolbar: screen only, never printed */}
   363→      <div className="flex flex-col gap-3 rounded-lg border border-border/80 bg-white p-3 sm:flex-row sm:items-center sm:justify-between print:hidden">
   364→        <div className="flex flex-wrap items-center gap-2">
   365→          <span className="text-xs font-medium text-muted-foreground">Student</span>
   366→          {/* ds-allow-native-select: shadcn Select is a Radix listbox trigger, not a native select */}
   367→          <Select value={student.id} onValueChange={switchStudent}>
   368→            <SelectTrigger className="h-8 w-full text-compact sm:w-65">
   369→              <SelectValue />
   370→            </SelectTrigger>
   371→            <SelectContent>
   372→              {classmates.map((s) => (
   373→                <SelectItem key={s.id} value={s.id}>
   374→                  {s.name}
   375→                </SelectItem>
   376→              ))}
   377→            </SelectContent>
   378→          </Select>
   379→        </div>
   380→        <div className="flex items-center gap-2">
   381→          <Button size="sm" className="gap-1.5" onClick={downloadPdf}>
   382→            <Download className="size-4" /> Download PDF
   383→          </Button>
   384→          <Button size="sm" variant="outline" className="gap-1.5" onClick={() => window.print()}>
   385→            <Printer className="size-4" /> Print
   386→          </Button>
   387→        </div>
   388→      </div>
   389→
   390→      {/* Legacy rule 7: a student with no marks at all renders an empty
   391→          subject table — warn before printing instead of generating. */}
   392→      {assessed.length === 0 ? (
   393→        <div className="flex items-center gap-2 rounded-lg border border-destructive/40 bg-destructive/8 px-3 py-2 text-compact text-destructive print:hidden">
   394→          <AlertTriangle className="size-4 shrink-0" />
   395→          No marks have been entered for this student yet — report cards are generated only after marks exist.
   396→        </div>
   397→      ) : null}
   398→
   399→      {/* The printable card: scrollable on a phone, exact A4 sheet in print.
   400→          Totally white paper; every section is a bordered table. */}
   401→      <div className="overflow-x-auto print:overflow-visible">
   402→        <div className="print-area mx-auto w-full max-w-205 border border-border bg-white p-8 font-serif text-foreground shadow-sm">
   403→          {/* School header: logo + school block on white */}
   404→          <div className="flex items-center gap-4">
   405→            <img
   406→              src="/josephian-logo.png"
   407→              alt="Josephian Schools logo"
   408→              className="h-20 w-20 shrink-0"
   409→            />
   410→            <div className="min-w-0 flex-1 text-center">
   411→              <p
   412→                className="text-lg font-bold uppercase tracking-[0.2em]"
   413→                style={{ color: BRAND }}
   414→              >
   415→                {SCHOOL.name}
   416→              </p>
   417→              <p className="mt-0.5 text-xs italic text-muted-foreground">{SCHOOL.motto}</p>
   418→              <p className="mt-0.5 text-2xs text-muted-foreground">
   419→                {SCHOOL.address} · {SCHOOL.phone} · {SCHOOL.email}
   420→              </p>
   421→            </div>
   422→            <div aria-hidden className="h-20 w-20 shrink-0" />
   423→          </div>
   424→
   425→          {/* Title band — maroon table header row */}
   426→          <table className={cn(TBL, "mt-4")}>
   427→            <tbody>
   428→              <tr>
   429→                <th scope="colgroup" className="bg-[#5C0A14] px-3 py-2 text-center">
   430→                  <span className="block text-sm font-bold uppercase tracking-[0.3em] text-white">
   431→                    Student Report Card
   432→                  </span>
   433→                  <span className="mt-0.5 block text-2xs font-medium tracking-[0.1em] text-white/85">
   434→                    {cycle.term} · {cycle.academicYear} Academic Year · {cycle.name}
   435→                  </span>
   436→                </th>
   437→              </tr>
   438→            </tbody>
   439→          </table>
   440→
   441→          {/* Student particulars */}
   442→          <table className={cn(TBL, "mt-4")}>
   443→            <colgroup>
   444→              <col className="w-[17%]" />
   445→              <col className="w-[33%]" />
   446→              <col className="w-[17%]" />
   447→              <col className="w-[33%]" />
   448→            </colgroup>
   449→            <tbody>
   450→              <tr>
   451→                <th colSpan={4} scope="colgroup" className={SECTION_BAR}>
   452→                  Student Particulars
   453→                </th>
   454→              </tr>
   455→              <tr>
   456→                <th scope="row" className={cn(CELL, "text-left", CELL_LABEL)}>
   457→                  Student name
   458→                </th>
   459→                <td className={cn(CELL, "font-medium")}>{student.name}</td>
   460→                <th scope="row" className={cn(CELL, "text-left", CELL_LABEL)}>
   461→                  Admission no
   462→                </th>
   463→                <td className={cn(CELL, "font-medium tabular")}>{student.admissionNo}</td>
   464→              </tr>
   465→              <tr>
   466→                <th scope="row" className={cn(CELL, "text-left", CELL_LABEL)}>
   467→                  Class
   468→                </th>
   469→                <td className={cn(CELL, "font-medium")}>{classNameOf(student.classId)}</td>
   470→                <th scope="row" className={cn(CELL, "text-left", CELL_LABEL)}>
   471→                  Class teacher
   472→                </th>
   473→                <td className={cn(CELL, "font-medium")}>{classTeacherName}</td>
   474→              </tr>
   475→              <tr>
   476→                <th scope="row" className={cn(CELL, "text-left", CELL_LABEL)}>
   477→                  Academic year
   478→                </th>
   479→                <td className={cn(CELL, "font-medium tabular")}>{cycle.academicYear}</td>
   480→                <th scope="row" className={cn(CELL, "text-left", CELL_LABEL)}>
   481→                  Report cycle
   482→                </th>
   483→                <td className={cn(CELL, "font-medium")}>{cycle.name}</td>
   484→              </tr>
   485→              <tr>
   486→                <th scope="row" className={cn(CELL, "text-left", CELL_LABEL)}>
   487→                  Report date
   488→                </th>
   489→                <td className={cn(CELL, "font-medium tabular")}>
   490→                  {fmtLongDate(cycle.reportDate)}
   491→                </td>
   492→                <th scope="row" className={cn(CELL, "text-left", CELL_LABEL)}>
   493→                  School opens
   494→                </th>
   495→                <td className={cn(CELL, "font-medium tabular")}>
   496→                  {fmtLongDate(cycle.schoolOpensDate)}
   497→                </td>
   498→              </tr>
   499→            </tbody>
   500→          </table>
   501→
   502→          {/* Subject results — the card's core table: monthly and terminal
   503→              components with their assessment weights, weighted total, grade
   504→              badge and exam-vs-CA progress, with the subject teacher's remark
