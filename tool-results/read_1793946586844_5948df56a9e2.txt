    56→/**
    57→ * The card is printed as pure white paper laid out entirely with bordered
    58→ * tables (the school's requested "totally white, only table" design). The
    59→ * single accent colour is the brand maroon #5C0A14 — the exact maroon of
    60→ * the Josephian Schools shield — used for the title band and every table
    61→ * header row; all other fills are white.
    62→ */
    63→const BRAND = "#5C0A14";
    64→
    65→/* Shared chrome for the card's tables. */
    66→const TBL = "w-full table-fixed border-collapse";
    67→/** Full-width maroon section bar (first row of each section table). */
    68→const SECTION_BAR =
    69→  "border border-border bg-[#5C0A14] px-3 py-1.5 text-center text-[11px] font-bold uppercase tracking-[0.22em] text-white";
    70→/** Maroon column-heading cell. */
    71→const COL_HEAD =
    72→  "border border-border bg-[#5C0A14] px-2 py-1.5 text-[11px] font-bold uppercase tracking-[0.1em] text-white";
    73→/** Standard bordered body cell (white). */
    74→const CELL = "border border-border bg-white px-2 py-1.5 align-middle";
    75→/** Tiny uppercase label used inside bordered cells. */
    76→const CELL_LABEL =
    77→  "text-3xs font-semibold uppercase tracking-[0.12em] text-muted-foreground";
    78→
    79→interface ChartRow {
    80→  name: string;
    81→  achieved: number;
    82→  classAvg: number;
    83→  classTop: number;
    84→}
    85→
    86→interface PerformanceChartSectionProps {
    87→  /** Which report engine printed this card (mid-term vs terminal rules). */
    88→  variant: ChartVariant;
    89→  /** Static data-URI chart image; null renders the fallback paragraph. */
    90→  chartImage: string | null;
    91→  rows: ChartRow[];
    92→  strongest: { name: string; score: number } | null;
    93→  /** Mid-Term primary cards only (chart prompt §8). */
    94→  bestImprovement: { name: string; value: number } | null;
    95→  isPrimary: boolean;
    96→}
    97→
    98→/**
    99→ * "Performance Charts" section (A4 section 6 / chart prompt §8) — also laid
   100→ * out as a table: a maroon section bar and a single bordered body cell
   101→ * holding the badge lines, the static grouped bar chart image, the Terminal
   102→ * HTML legend and a visually hidden data table backing the chart for screen
   103→ * readers (B0: charts include a table fallback).
   104→ */
   105→function PerformanceChartSection({
   106→  variant,
   107→  chartImage,
   108→  rows,
   109→  strongest,
   110→  bestImprovement,
   111→  isPrimary,
   112→}: PerformanceChartSectionProps) {
   113→  // Section heading: "Performance Chart" on secondary terminal cards only.
   114→  const heading = variant === "mid-term" || isPrimary ? "Performance Charts" : "Performance Chart";
   115→  return (
   116→    <table className={cn(TBL, "mt-4")}>
   117→      <tbody>
   118→        <tr>
   119→          <th scope="colgroup" className={SECTION_BAR}>
   120→            {heading}
   121→          </th>
   122→        </tr>
   123→        <tr>
   124→          <td className={CELL}>
   125→            {strongest || bestImprovement ? (
   126→              <div className="mb-1.5 flex flex-wrap gap-x-4 gap-y-0.5 text-2xs font-semibold">
   127→                {strongest ? (
   128→                  <span style={{ color: BRAND }}>
   129→                    ★ Strongest Subject: {strongest.name} — {fmtTotalMark(strongest.score)} / 100
   130→                  </span>
   131→                ) : null}
   132→                {bestImprovement ? (
   133→                  <span className="text-[#2D6A4E]">
   134→                    ▲ Best Improvement: {bestImprovement.name} — +{bestImprovement.value}
   135→                  </span>
   136→                ) : null}
   137→              </div>
   138→            ) : null}
   139→            {chartImage ? (
   140→              <img
   141→                src={chartImage}
   142→                alt="Grouped bar chart comparing achieved marks with the class average and class topper for every assessed subject"
   143→                className="chart-img w-full"
   144→              />
   145→            ) : (
   146→              <p className="text-2xs text-muted-foreground">{CHART_FALLBACK_TEXT}</p>
   147→            )}
   148→            {variant === "terminal" ? (
   149→              // Terminal legend is rendered in HTML, not in the chart image (§9).
   150→              <div className="mt-1 flex flex-wrap items-center justify-center gap-x-4 gap-y-1 text-2xs text-muted-foreground">
   151→                <span className="inline-flex items-center gap-1">
   152→                  <span className="size-2 rounded-sm" style={{ background: "#E74C3C" }} /> Achieved Marks
   153→                </span>
   154→                <span className="inline-flex items-center gap-1">
   155→                  <span className="size-2 rounded-sm" style={{ background: "#3498DB" }} /> Class Avg
   156→                </span>
   157→                <span className="inline-flex items-center gap-1">
   158→                  <span className="size-2 rounded-sm" style={{ background: "#2ECC71" }} /> Class Topper
   159→                </span>
   160→              </div>
   161→            ) : null}
   162→            {/* Chart data-table fallback (B0: charts include a table fallback). */}
   163→            <table className="sr-only">
   164→              <caption>Subject totals against the class average and class top</caption>
   165→              <thead>
   166→                <tr>
   167→                  <th scope="col">Subject</th>
   168→                  <th scope="col">Achieved</th>
   169→                  <th scope="col">Class average</th>
   170→                  <th scope="col">Class top</th>
   171→                </tr>
   172→              </thead>
   173→              <tbody>
   174→                {rows.map((row) => (
   175→                  <tr key={row.name}>
   176→                    <th scope="row">{row.name}</th>
   177→                    <td>{fmtTotalMark(row.achieved)}</td>
   178→                    <td>{fmtTotalMark(row.classAvg)}</td>
   179→                    <td>{fmtTotalMark(row.classTop)}</td>
   180→                  </tr>
   181→                ))}
   182→              </tbody>
   183→            </table>
   184→          </td>
   185→        </tr>
   186→      </tbody>
   187→    </table>
   188→  );
   189→}

   191→export default function ReportCardPage() {
   192→  const params = useParams<{ studentId?: string }>();
   193→  const router = useRouter();
   194→  const { cycle } = useCycleContext();
   195→  const student = studentById(params.studentId ?? "") ?? studentById(DEMO_IDS.studentId)!;
   196→
   197→  const result = studentResultOf(student.id);
   198→  const classmates = studentsInClass(student.classId);
   199→  const classSize = classmates.length;
   200→  const attendance = attendanceRecordFor(student.id);
   201→  const comment = commentFor(student.classId, student.id);
   202→  const classTeacherName =
   203→    teacherById(classById(student.classId)?.classTeacherId)?.name ?? "\u2014";
   204→
   205→  const classSubjects = classSubjectsFor(student.classId);
   206→  const scheme = schemeForClass(student.classId);
   207→  // ── Two report card templates ────────────────────────────────────────
   208→  // The student's school level selects the template, so choosing a class
   209→  // in the toolbar or the header switches between them:
   210→  //   • PRIMARY (Grades 1–7) — A–E bands, no points, no division.
   211→  //   • SECONDARY (Forms 1–4) — NECTA CSEE A–F bands, best-7 points,
   212→  //     Division and the Division Scale strip.
   213→  // `showPoints` below gates every secondary-only section; primary cards
   214→  // simply never render points or division anywhere.
   215→  const isPrimary = isPrimaryClass(student.classId);
   216→  // Assessment split shown in the table header (25 % monthly / 75 % terminal).
   217→  const caPct = classSubjects[0]?.caMarks ?? 25;
   218→  const examPct = classSubjects[0]?.examMarks ?? 75;
   219→  // COMBINED cycles (two exam components, e.g. Monthly + Mid-Term) label the
   220→  // two mark columns after their own exams and weights; plain cycles keep
   221→  // the classic Monthly / Terminal captions.
   222→  const combined =
   223→    cycle.components && cycle.components.length === 2 ? cycle.components : null;
   224→  const firstCol = { label: combined?.[0].label ?? "Monthly", pct: combined?.[0].weight ?? caPct };
   225→  const secondCol = { label: combined?.[1].label ?? "Terminal", pct: combined?.[1].weight ?? examPct };
   226→  // Top band of the class scheme — rendered as the filled grade badge.
   227→  const topGrade = scheme.boundaries[0]?.grade;
   228→  const allRows = classSubjects.map((cs) => {
   229→    const subject = subjectById(cs.subjectId)!;
   230→    const comp = getComponents(student.classId, cs.subjectId)[student.id];
   231→    const enrolled = comp?.enrolled ?? true;
   232→    const mark = enrolled ? result?.marks[cs.subjectId] ?? null : null;
   233→    const grade = mark !== null ? gradeFor(mark, scheme) : undefined;
   234→    return {
   235→      subject,
   236→      enrolled,
   237→      mark,
   238→      grade,
   239→      ca: comp?.ca ?? null,
   240→      exam: comp?.exam ?? null,
   241→    };
   242→  });
   243→  // A subject with no mark at all (both components ABS/NA/NR/PENDING) is
   244→  // omitted from the card entirely; non-enrolled subsidiaries stay with "—".
   245→  // Remaining rows renumber continuously after any omitted subject (the #
   246→  // column below).
   247→  const rows = allRows.filter((r) => !r.enrolled || r.mark !== null);
   248→  const assessed = allRows.filter((r) => r.mark !== null);
   249→  const pointsTotal = assessed.reduce((a, r) => a + (r.grade?.points ?? 0), 0);
   250→  // Secondary division: best 7 subject points; null when fewer than 7 graded
   251→  // (and always null for primary classes — divisions are a NECTA outcome).
   252→  const division = isPrimary ? null : divisionFor(assessed.map((r) => r.grade?.points ?? null));
   253→  const passed = subjectsPassed(assessed.map((r) => r.grade?.grade ?? null));
   254→  // Strongest subject: highest total; ties go to the first in subject order.
   255→  const strongest =
   256→    assessed.length > 0
   257→      ? assessed.reduce((best, r) => ((r.mark as number) > (best.mark as number) ? r : best))
   258→      : undefined;
   259→  // Points and division are hidden on primary cards (Primary A–E bands carry
   260→  // no points) and on any secondary class whose teacher set the Report
   261→  // Overview "No Division" flag to Yes.
   262→  const noDivision = !isPrimary && (classById(student.classId)?.noDivision ?? false);
   263→  const showPoints = !isPrimary && !noDivision;
   264→  // Most improved (primary only): largest CA delta C2 − C1 per subject; a
   265→  // single recorded term counts as 0. Ties go to the first subject in order.
   266→  const mostImproved = (() => {
   267→    if (!isPrimary) return null;
   268→    const deltas = classSubjects.map((cs) => {
   269→      const comp = getComponents(student.classId, cs.subjectId)[student.id];
   270→      const cur = comp?.enrolled ? comp.ca ?? null : null;
   271→      const prev = prevCaOf(student.classId, cs.subjectId)[student.id] ?? null;
   272→      return { subjectId: cs.subjectId, delta: cur !== null && prev !== null ? cur - prev : 0 };
   273→    });
   274→    const best = mostImprovedSubject(deltas);
   275→    return best && best.delta > 0 ? best : null;
   276→  })();
   277→  const mostImprovedSubjectRow = mostImproved
   278→    ? rows.find((r) => r.subject.id === mostImproved.subjectId)
   279→    : undefined;
   280→
   281→  // Chart data (A4 section 6 / chart prompt §3–§4): achieved total vs class
   282→  // average and class top per assessed subject, over the student's enrolled
   283→  // classmates. Class stats follow the cycle's engine: mid-term counts a
   284→  // student with a single assessment at 100%; terminal only counts students
   285→  // with BOTH components (chart prompt §4/§9).
   286→  // Monthly Test and Mid-Term cycles run the mid-term chart engine (a
   287→  // single assessment carries at 100%); terminal and annual run terminal.
   288→  const variant: ChartVariant =
   289→    cycle.kind === "terminal" || cycle.kind === "annual" ? "terminal" : "mid-term";
   290→  // Terminal and annual cards carry the subject teacher's remark as a
   291→  // full-width row beneath each subject, with no caption (school rule):
   292→  // Mid-Term and Monthly templates have NO subject comments at all.
   293→  const isTerminalLike = cycle.kind === "terminal" || cycle.kind === "annual";
   294→  const chartRows: ChartRow[] = assessed.map((r) => {
   295→    const cs = classSubjects.find((c) => c.subjectId === r.subject.id);
   296→    const comps = getComponents(student.classId, r.subject.id);
   297→    const scores = Object.values(comps)
   298→      .filter((c) => c.enrolled)
   299→      .map((c) => classScoreFor(variant, c.ca, c.exam, cs?.caMarks ?? 25, cs?.examMarks ?? 75))
   300→      .filter((t): t is number => t !== null);
   301→    return {
   302→      name: r.subject.name,
   303→      achieved: r.mark ?? 0,
   304→      classAvg: scores.length ? scores.reduce((a, b) => a + b, 0) / scores.length : 0,
   305→      classTop: scores.length ? Math.max(...scores) : 0,
   306→    };
   307→  });
   308→  // Series + static chart image (chart prompt §7 call sequence).
   309→  const chartSeries = buildChartSeries(
   310→    chartRows.map((r) => ({ name: r.name, achieved: r.achieved })),
   311→    chartRows.map((r) => r.classAvg),
   312→    chartRows.map((r) => r.classTop),
   313→    strongest?.subject.name ?? "",
   314→    variant,
   315→  );
   316→  const chartImage = generateChartImage(chartSeries, variant);
   317→  // Best-Improvement badge: Mid-Term primary cards only (chart prompt §8).
   318→  const chartBestImprovement =
   319→    variant === "mid-term" && mostImproved && mostImprovedSubjectRow
   320→      ? { name: mostImprovedSubjectRow.subject.name, value: mostImproved.delta }
   321→      : null;
   322→
   323→  const switchStudent = (id: string) => {
   324→    setStoreStudent(id);
   325→    router.push(`/academic/report-cycles/${cycle.id}/students/${id}/report-card`);
   326→  };
   327→
   328→  // Class switch on the card itself: opens the class's first student, so
   329→  // the primary/secondary template follows the selected class.
   330→  const switchClass = (classId: string) => {
   331→    const first = studentsInClass(classId)[0];
   332→    if (!first) return;
   333→    setStoreClass(classId);
   334→    setStoreStudent(first.id);
   335→    router.push(`/academic/report-cycles/${cycle.id}/students/${first.id}/report-card`);
   336→  };
   337→
   338→  const downloadPdf = usePrintDownload(
   339→    "Use 'Save as PDF' in the print dialog to download the report card.",
   340→  );
   341→  const setStoreClass = useAppContext((s) => s.setClass);
   342→  const setStoreStudent = useAppContext((s) => s.setStudent);
   343→
   344→  // Performance summary cells; the points/division pair appears only where
   345→  // points are printed (secondary classes without the No-Division flag).
   346→  const summaryCells = [
   347→    { label: "Total marks", value: result ? fmtTotalMark(result.total) : "-", accent: false },
   348→    { label: "Average", value: result ? result.average.toFixed(1) : "-", accent: false },
   349→    {
   350→      label: "Overall grade",
   351→      value: result && result.count > 0 ? result.grade : "-",
   352→      accent: true,
   353→    },
   354→    ...(showPoints
   355→      ? [
   356→          {
   357→            label: "Points total",
   358→            value: assessed.length > 0 ? String(pointsTotal) : "-",
   359→            accent: false,
   360→          },
   361→          {
   362→            label: "Division",
   363→            value: division ? `Div ${division.division}` : "—",
   364→            accent: true,
   365→          },
   366→        ]
   367→      : []),
   368→    {
   369→      label: "Subjects passed",
   370→      value: assessed.length > 0 ? `${passed} of ${assessed.length}` : "-",
   371→      accent: false,
   372→    },
   373→    { label: "Class size", value: String(classSize), accent: false },
   374→  ];
   375→
   376→  // Attendance row cells; "Not recorded" appears only when > 0 (A3.8).
   377→  const rateText = attendance.rate === null ? "–" : `${attendance.rate.toFixed(1)}%`;
   378→  const attRatingText =
   379→    attendance.rating === "none" ? null : ATTENDANCE_RATING_TEXT[attendance.rating];
   380→  const attCells = [
   381→    {
   382→      label: "Days attended / recorded",
   383→      value: `${attendance.daysAttended} / ${attendance.daysRecorded}`,
   384→    },
   385→    { label: "Attendance rate", value: rateText },
   386→    { label: "Late", value: String(attendance.late) },
   387→    { label: "Absent", value: String(attendance.absent) },
   388→    { label: "Excused", value: String(attendance.excused) },
   389→    ...(attendance.notRecorded > 0
   390→      ? [
   391→          {
   392→            label: "Not recorded",
   393→            value: `${attendance.notRecorded} day${attendance.notRecorded === 1 ? "" : "s"}`,
   394→          },
   395→        ]
   396→      : []),
   397→  ];
   398→
   399→  return (
   400→    <div className="space-y-5">
   401→      <PageHeader
   402→        title="Report card"
   403→        description={`${student.name} · ${classNameOf(student.classId)} · ${cycle.name}`}
   404→      />
   405→
   406→      {/* Toolbar: screen only, never printed */}
   407→      <div className="flex flex-col gap-3 rounded-lg border border-border/80 bg-white p-3 sm:flex-row sm:items-center sm:justify-between print:hidden">
   408→        <div className="flex flex-wrap items-center gap-2">
   409→          <span className="text-xs font-medium text-muted-foreground">Class</span>
   410→          {/* ds-allow-native-select: shadcn Select is a Radix listbox trigger, not a native select */}
   411→          <Select value={student.classId} onValueChange={switchClass}>
   412→            <SelectTrigger className="h-8 w-full text-compact sm:w-45" aria-label="Class">
   413→              <SelectValue />
   414→            </SelectTrigger>
   415→            <SelectContent>
   416→              {classes.map((c) => (
   417→                <SelectItem key={c.id} value={c.id}>
   418→                  {c.name}
   419→                </SelectItem>
   420→              ))}
   421→            </SelectContent>
   422→          </Select>
   423→          <span className="text-xs font-medium text-muted-foreground">Student</span>
   424→          {/* ds-allow-native-select: shadcn Select is a Radix listbox trigger, not a native select */}
   425→          <Select value={student.id} onValueChange={switchStudent}>
   426→            <SelectTrigger className="h-8 w-full text-compact sm:w-65" aria-label="Student">
   427→              <SelectValue />
   428→            </SelectTrigger>
   429→            <SelectContent>
   430→              {classmates.map((s) => (
   431→                <SelectItem key={s.id} value={s.id}>
   432→                  {s.name}
   433→                </SelectItem>
   434→              ))}
   435→            </SelectContent>
   436→          </Select>
   437→        </div>
   438→        <div className="flex items-center gap-2">
   439→          <Button size="sm" className="gap-1.5" onClick={downloadPdf}>
   440→            <Download className="size-4" /> Download PDF
   441→          </Button>
   442→          <Button size="sm" variant="outline" className="gap-1.5" onClick={() => window.print()}>
   443→            <Printer className="size-4" /> Print
   444→          </Button>
   445→        </div>
   446→      </div>
   447→
   448→      {/* Legacy rule 7: a student with no marks at all renders an empty
   449→          subject table — warn before printing instead of generating. */}
   450→      {assessed.length === 0 ? (
   451→        <div className="flex items-center gap-2 rounded-lg border border-destructive/40 bg-destructive/8 px-3 py-2 text-compact text-destructive print:hidden">
   452→          <AlertTriangle className="size-4 shrink-0" />
   453→          No marks have been entered for this student yet — report cards are generated only after marks exist.
   454→        </div>
   455→      ) : null}
   456→
   457→      {/* The printable card: scrollable on a phone, exact A4 sheet in print.
   458→          Totally white paper; every section is a bordered table. */}
   459→      <div className="overflow-x-auto print:overflow-visible">
   460→        <div className="print-area mx-auto w-full max-w-205 border border-border bg-white p-8 font-serif text-foreground shadow-sm">
   461→          {/* School header: logo + school block on white */}
   462→          <div className="flex items-center gap-4">
   463→            <img
   464→              src="/josephian-logo.png"
   465→              alt="Josephian Schools logo"
   466→              className="h-20 w-20 shrink-0"
   467→            />
   468→            <div className="min-w-0 flex-1 text-center">
   469→              <p
   470→                className="text-lg font-bold uppercase tracking-[0.2em]"
   471→                style={{ color: BRAND }}
   472→              >
   473→                {SCHOOL.name}
   474→              </p>
   475→              <p className="mt-0.5 text-xs italic text-muted-foreground">{SCHOOL.motto}</p>
   476→              <p className="mt-0.5 text-2xs text-muted-foreground">
   477→                {SCHOOL.address} · {SCHOOL.phone} · {SCHOOL.email}
   478→              </p>
   479→            </div>
   480→            <div aria-hidden className="h-20 w-20 shrink-0" />
   481→          </div>
   482→
   483→          {/* Title band — maroon table header row */}
   484→          <table className={cn(TBL, "mt-4")}>
   485→            <tbody>
   486→              <tr>
   487→                <th scope="colgroup" className="bg-[#5C0A14] px-3 py-2 text-center">
   488→                  <span className="block text-sm font-bold uppercase tracking-[0.3em] text-white">
   489→                    Student Report Card
   490→                  </span>
   491→                  <span className="mt-0.5 block text-2xs font-medium tracking-[0.1em] text-white/85">
   492→                    {cycle.term} · {cycle.academicYear} Academic Year · {cycle.name}
   493→                  </span>
   494→                </th>
   495→              </tr>
   496→            </tbody>
   497→          </table>
